## 2. 实验目标

1. 能标定 4~6 个航点并写成 YAML（名称、坐标、偏航角、停留时长），坐标经 RViz 与卷尺双向复核。
2. 能用 `nav2_msgs/action/NavigateToPose` 客户端按列表顺序发点，逐点等待结果而不并发抢目标。
3. 能实现逐点停留、单点失败重试（默认 2 次）、连续 3 点失败即整体中止并告警的异常策略。
4. 能实现 `Ctrl+C` 立即取消当前目标，并通过 `start_index` 参数从断点续跑。

### 5.2 写入完整脚本

把下面全文写入 `~/yhs_labs/S2-10/patrol_loop.py`：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""S2-10 夜间库区盘点巡航：NavigateToPose 顺序发点，含停留、重试、取消与断点续跑。"""

import math
import os
import sys

import rclpy
import yaml
from action_msgs.msg import GoalStatus
from geometry_msgs.msg import PoseStamped
from nav2_msgs.action import NavigateToPose
from rclpy.action import ActionClient
from rclpy.node import Node

RED, YEL, GRN, RST = '\033[31m', '\033[33m', '\033[32m', '\033[0m'
STATUS = {GoalStatus.STATUS_SUCCEEDED: 'SUCCEEDED', GoalStatus.STATUS_CANCELED: 'CANCELED',
          GoalStatus.STATUS_ABORTED: 'ABORTED'}


class PatrolLoop(Node):
    def __init__(self):
        super().__init__('patrol_loop_s2_10')
        for name, value in (('route_file', '~/yhs_labs/S2-10/waypoints.yaml'),
                            ('frame_id', 'map'), ('loops', 4), ('dwell', 5.0),
                            ('max_retries', 2), ('fail_limit', 3), ('start_index', 0),
                            ('goal_timeout', 180.0), ('retry_wait', 10.0)):
            self.declare_parameter(name, value)
        self.route_file = os.path.expanduser(str(self.get_parameter('route_file').value))
        self.frame_id = str(self.get_parameter('frame_id').value)
        self.loops = int(self.get_parameter('loops').value)
        self.dwell = float(self.get_parameter('dwell').value)
        self.max_retries = int(self.get_parameter('max_retries').value)
        self.fail_limit = int(self.get_parameter('fail_limit').value)
        self.start_index = int(self.get_parameter('start_index').value)
        self.timeout = float(self.get_parameter('goal_timeout').value)
        self.retry_wait = float(self.get_parameter('retry_wait').value)
        self.waypoints = self.load_route()
        self.client = ActionClient(self, NavigateToPose, 'navigate_to_pose')
        self.goal_handle, self.canceled = None, False
        self.current_name, self.last_fb = '', 0.0
        self.stats = {'ok': 0, 'fail': 0, 'retry': 0}

    def load_route(self):
        with open(self.route_file, 'r', encoding='utf-8') as f:
            doc = yaml.safe_load(f)
        self.frame_id = str(doc.get('frame_id', self.frame_id))
        wps = doc.get('waypoints') or []
        if not wps:
            raise RuntimeError(f'{self.route_file} 中没有 waypoints')
        self.get_logger().info(f'已加载 {len(wps)} 个航点')
        return wps

    def now_s(self):
        return self.get_clock().now().nanoseconds / 1e9

    def spin_for(self, seconds):
        end = self.now_s() + seconds
        while rclpy.ok() and not self.canceled and self.now_s() < end:
            rclpy.spin_once(self, timeout_sec=0.1)

    def wait(self, future, timeout):
        rclpy.spin_until_future_complete(self, future, timeout_sec=timeout)
        return future.result() if future.done() else None

    def on_feedback(self, msg):
        now = self.now_s()
        if now - self.last_fb < 2.0:
            return
        self.last_fb = now
        fb = msg.feedback
        print(f'  {self.current_name} 剩余 {fb.distance_remaining:.2f} m，'
              f'恢复 {fb.number_of_recoveries} 次', flush=True)

    def send_goal(self, wp):
        ps = PoseStamped()
        ps.header.frame_id = self.frame_id
        ps.header.stamp = self.get_clock().now().to_msg()
        ps.pose.position.x = float(wp['x'])
        ps.pose.position.y = float(wp['y'])
        yaw = float(wp.get('yaw', 0.0))
        ps.pose.orientation.z = math.sin(yaw / 2.0)
        ps.pose.orientation.w = math.cos(yaw / 2.0)
        goal = NavigateToPose.Goal()
        goal.pose = ps
        gh = self.wait(self.client.send_goal_async(goal, feedback_callback=self.on_feedback), 10.0)
        if gh is None or not gh.accepted:
            self.get_logger().error(f'{wp["name"]} 目标被拒绝')
            return GoalStatus.STATUS_ABORTED, 0.0
        self.goal_handle = gh
        t0 = self.now_s()
        res = self.wait(gh.get_result_async(), self.timeout)
        elapsed = self.now_s() - t0
        if res is None:
            self.get_logger().error(f'{wp["name"]} 超时 {self.timeout:.0f} s，主动取消')
            self.cancel_current('单点超时')
            return GoalStatus.STATUS_ABORTED, elapsed
        return res.status, elapsed

    def cancel_current(self, reason=''):
        if self.goal_handle is None:
            return
        self.wait(self.goal_handle.cancel_goal_async(), 5.0)
        self.goal_handle = None
        print(YEL + f'[取消] 当前目标已取消（{reason or "人工中断"}）' + RST, flush=True)

    def run(self):
        if not self.client.wait_for_server(timeout_sec=10.0):
            self.get_logger().error('navigate_to_pose 不存在，请确认 Nav2 已 active')
            return False
        consecutive = 0
        for lap in range(1, self.loops + 1):
            first = self.start_index if lap == 1 else 0
            print(f'--- 第 {lap}/{self.loops} 圈，从第 {first + 1} 个航点开始 ---', flush=True)
            for idx in range(first, len(self.waypoints)):
                if self.canceled:
                    return False
                wp = self.waypoints[idx]
                self.current_name = wp['name']
                ok, status, elapsed = False, None, 0.0
                for attempt in range(self.max_retries + 1):
                    status, elapsed = self.send_goal(wp)
                    if self.canceled:
                        return False
                    if status == GoalStatus.STATUS_SUCCEEDED:
                        ok = True
                        break
                    self.stats['retry'] += 1
                    print(YEL + f'[重试] {wp["name"]} 第 {attempt + 1} 次失败'
                          f'（{STATUS.get(status, status)}），{self.retry_wait:.0f} s 后重试'
                          + RST, flush=True)
                    self.spin_for(self.retry_wait)
                if ok:
                    consecutive = 0
                    self.stats['ok'] += 1
                    print(GRN + f'[到达] 圈 {lap}/{self.loops} 航点 {wp["name"]} '
                          f'({float(wp["x"]):.2f}, {float(wp["y"]):.2f}) 用时 {elapsed:.1f} s'
                          + RST, flush=True)
                    self.spin_for(float(wp.get('dwell', self.dwell)))
                else:
                    consecutive += 1
                    self.stats['fail'] += 1
                    print(RED + f'[失败] {wp["name"]} 连续失败 {consecutive} 个航点' + RST,
                          flush=True)
                    if consecutive >= self.fail_limit:
                        self.get_logger().error(f'连续 {self.fail_limit} 个航点失败，中止巡航')
                        print(RED + '[中止] 请人工检查现场后重跑' + RST, flush=True)
                        return False
        return True


def main(args=None):
    rclpy.init(args=args)
    node = PatrolLoop()
    ok = False
    try:
        ok = node.run()
    except KeyboardInterrupt:
        print(YEL + '[中断] Ctrl+C，取消当前目标' + RST, flush=True)
        node.canceled = True
        node.cancel_current('Ctrl+C')
    finally:
        s = node.stats
        node.get_logger().info(f'巡航结束：到达 {s["ok"]}，失败 {s["fail"]}，重试 {s["retry"]}')
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()
    sys.exit(0 if ok else 1)


if __name__ == '__main__':
    main()
```

## 6. 关键代码与配置

| 代码段 | 作用 | 关键点 |
|---|---|---|
| `load_route()` | 读 YAML 航点 | 缺 `waypoints` 立即抛错，避免空跑 |
| `send_goal()` | 串行发点 | 单目标阻塞等待，避免多目标抢占导致行为树频繁 preempt |
| `wait()` | 统一超时等待 | 超时返回 `None`，由调用方取消，绝不死等 |
| `cancel_current()` | 标准取消通道 | `goal_handle.cancel_goal_async()`，不用伪造服务名 |
| `spin_for()` | 停留/重试间隔 | 用 `spin_once` 循环，等待期间 feedback 与信号仍被处理 |
| `fail_limit` | 连续失败保护 | 达到 3 个即返回 `False`，上层据此通知值班员 |
| `start_index` | 断点续跑 | 只作用于第 1 圈，后续圈从 0 开始 |

| 接口 | 类型 | 方向 | 说明 |
|---|---|---|---|
| `/navigate_to_pose` | `nav2_msgs/action/NavigateToPose` | 客户端 | 逐点导航，feedback 含 `distance_remaining`、`number_of_recoveries` |
| `/amcl_pose` | `geometry_msgs/msg/PoseWithCovarianceStamped` | 只读 | 航点坐标来源 |

| 参数 | 默认 | 说明 |
|---|---|---|
| `route_file` | `~/yhs_labs/S2-10/waypoints.yaml` | 航点文件 |
| `loops` / `dwell` | 4 / 5.0 | 圈数 / 默认停留（s），可被单点 `dwell` 覆盖 |
| `max_retries` / `fail_limit` | 2 / 3 | 单点额外重试次数 / 连续失败上限 |
| `start_index` | 0 | 断点续跑起始下标（0 基） |
| `goal_timeout` / `retry_wait` | 180.0 / 10.0 | 单点等待上限 / 重试前等待（s） |

## 10. 回滚与常见问题

回滚：①`Ctrl+C` 取消当前目标并退出；②确认 `/cmd_vel` 已归零、车辆静止；③`rm -rf ~/yhs_labs/S2-10`；④`git status` 应无新增；⑤本实验不改地图与 Nav2 参数，现场恢复照明与人行通道即可。

| 现象 | 原因 | 处理 |
|---|---|---|
| `navigate_to_pose 不存在` | Nav2 未 active 或命名空间不同 | `ros2 lifecycle get /bt_navigator`；确认未加 `__ns` |
| 目标被拒绝 | 航点在未知区域或膨胀区内 | RViz 检查 `global_costmap`；把航点移出障碍 0.3 m |
| 连续重试后中止 | 通道被堵、定位跳变 | 现场排障；先按 S1-06 确认恢复行为生效 |
| 单圈超 12 min | 限速未收紧或路径绕远 | 核对 `max_vel_x` 与航点顺序 |
| `Ctrl+C` 后车缓慢停下 | `velocity_smoother` 减速限幅 | 属预期；需要立即停按硬件急停 |
| `ModuleNotFoundError: nav2_msgs` | 未 source ROS 2 | `source /opt/ros/humble/setup.bash` |
| 巡航中定位跳变导致偏航 | 长走廊激光退化 | 先按 S1-05 收紧 AMCL 参数，再复测 |
