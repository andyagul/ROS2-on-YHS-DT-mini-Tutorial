# S2-05 急停与手柄事件留痕

> 场景来源：[二次开发场景实施方案.md(../二次开发场景实施方案.md) S2-05；清单条目：[清单.md(../二次开发项目清单.md) 第二阶段 #5。
> 软硬件：DT-mini 底盘（急停、手柄接管反馈）；Ubuntu 22.04 + ROS 2 Humble。安全前提：急停演练由安全员执行，车辆架空或驻车；本脚本**只读**，不得接入急停控制链路。

## 2. 实验目标

1. 能说明 `io_fb.io_fb_estop`、`io_fb.io_fb_joypad_ctrl` 的触发极性，并区分"状态"与"事件"。
2. 能实现边沿检测：状态变化只记一次，避免高频反馈导致日志刷屏。
3. 能把时间戳、事件名、前后状态、上一状态持续时长、电压、SOC、档位写成 8 列 CSV。
4. 能实现 10 MB 单文件上限与历史文件轮转，8 h 运行不丢行。
5. 能完成 10 次急停 + 5 次手柄接管演练，事件数与日志行数一一对应，时间戳误差 ≤ 1 s。

## 5. 实施步骤

### 5.1 写入完整脚本

把下面全文写入 `~/yhs_labs/S2-05/safety_log.py`：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""S2-05 急停与手柄接管留痕：边沿检测 + CSV 日志 + 10 MB 轮转。"""

import csv
import os
from datetime import datetime

import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from yhs_can_interfaces.msg import ChassisInfoFb

HEADER = ['time', 'event', 'from', 'to', 'state_duration_s', 'voltage_v', 'soc_pct', 'gear']
SIGNALS = [('ESTOP', '急停', 'io_fb_estop'), ('JOYPAD', '手柄接管', 'io_fb_joypad_ctrl')]


class SafetyLog(Node):
    """状态变化即打印并落盘，同一状态只记一次。"""

    def __init__(self):
        super().__init__('safety_log_s2_05')
        params = (('log_dir', '~/yhs_labs/S2-05'), ('max_bytes', 10 * 1024 * 1024),
                  ('keep_files', 3), ('chassis_topic', '/chassis_info_fb'))
        for name, value in params:
            self.declare_parameter(name, value)
        log_dir = os.path.expanduser(str(self.get_parameter('log_dir').value))
        self.max_bytes = int(self.get_parameter('max_bytes').value)
        self.keep = int(self.get_parameter('keep_files').value)
        os.makedirs(log_dir, exist_ok=True)
        self.base = os.path.join(log_dir, 'safety_events.csv')
        self.file, self.writer = None, None
        self.open_log()
        self.prev = {}
        self.since = {}
        self.counts = {'ESTOP': 0, 'JOYPAD': 0}
        self.frames = 0
        self.msg = None
        qos = QoSProfile(depth=10, reliability=ReliabilityPolicy.RELIABLE,
                         history=HistoryPolicy.KEEP_LAST)
        self.sub = self.create_subscription(
            ChassisInfoFb, str(self.get_parameter('chassis_topic').value), self.on_chassis, qos)
        self.get_logger().info(f'事件留痕已启动：{self.base}，单文件上限 '
                               f'{self.max_bytes // 1048576} MB，保留 {self.keep} 个历史文件')
    def open_log(self):
        need_header = not os.path.exists(self.base) or os.path.getsize(self.base) == 0
        self.file = open(self.base, 'a', newline='', buffering=1)
        self.writer = csv.writer(self.file)
        if need_header:
            self.writer.writerow(HEADER)
    def rotate(self):
        self.file.close()
        for i in range(self.keep - 1, 0, -1):
            src, dst = f'{self.base}.{i}', f'{self.base}.{i + 1}'
            if os.path.exists(src):
                os.replace(src, dst)
        os.replace(self.base, f'{self.base}.1')
        self.open_log()
        self.get_logger().warn(f'日志达到上限，已轮转为 {self.base}.1')
    def now_s(self):
        return self.get_clock().now().nanoseconds / 1e9
    def on_chassis(self, msg):
        self.msg = msg
        self.frames += 1
        now = self.now_s()
        for key, label, field in SIGNALS:
            cur = bool(getattr(msg.io_fb, field))
            if key not in self.prev:
                self.prev[key], self.since[key] = cur, now
                self.get_logger().info(f'初始状态：{label} = {"触发/接管" if cur else "未触发"}')
                continue
            if cur == self.prev[key]:
                continue
            duration = now - self.since[key]
            self.write_row(key, label, self.prev[key], cur, duration)
            self.prev[key], self.since[key] = cur, now
    def write_row(self, key, label, old, cur, duration):
        stamp = datetime.now().strftime('%Y-%m-%d %H:%M:%S.%f')[:-3]
        v = self.msg.bms_fb.bms_fb_voltage
        soc = int(self.msg.bms_flag_fb.bms_flag_fb_soc)
        gear = self.msg.ctrl_fb.ctrl_fb_target_gear
        row = [stamp, key, '1' if old else '0', '1' if cur else '0',
               f'{duration:.3f}', f'{v:.2f}', str(soc), str(gear)]
        if self.file.tell() + len(','.join(row)) + 1 > self.max_bytes:
            self.rotate()
        self.writer.writerow(row)
        if cur:
            self.counts[key] += 1
        if key == 'ESTOP':
            arrow = '未触发 → 触发' if cur else '触发 → 释放'
        else:
            arrow = '人工 → 手柄' if cur else '手柄 → 人工'
        print(f'[{stamp}] {label} {arrow}  上一状态持续 {duration:.3f} s  '
              f'电压 {v:.2f} V  SOC {soc}%', flush=True)
    def shutdown(self):
        try:
            self.file.close()
        except Exception:
            pass
        self.get_logger().info(f'留痕退出：{self.frames} 帧，急停 {self.counts["ESTOP"]} 次，'
                               f'手柄 {self.counts["JOYPAD"]} 次')

def main(args=None):
    rclpy.init(args=args)
    node = SafetyLog()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.shutdown()
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()

if __name__ == '__main__':
    main()
```

## 6. 关键代码与配置

| 代码段 | 作用 | 关键点 |
|---|---|---|
| `SIGNALS` | 把事件名、中文名、`IoFb` 字段绑定 | 字段名与 `IoFb.msg` 完全一致 |
| `open_log` | 追加打开、按需写表头 | `newline=''` 避免 Windows 换行错位 |
| `on_chassis` | 边沿检测：`cur != prev` 才记录 | 高频反馈下不会刷屏；首次只记录初始态 |
| `write_row` | 写 8 列 CSV 并打印 | 持续时间用 `rclpy` 时钟差，精度到毫秒 |
| `rotate` | `keep_files` 份历史轮转 | 达到 10 MB 先轮转再写，单文件不超限 |
| `shutdown` | 关文件并汇总 | 统计为"触发边沿"次数，不含释放 |
| `QoSProfile` | RELIABLE + KEEP_LAST(10) | 与底盘 publisher（depth 1）兼容 |

| 字段 | 含义 | 示例 |
|---|---|---|
| `time` | 本机时间，毫秒精度 | `2026-11-16 21:03:11.482` |
| `event` | `ESTOP` / `JOYPAD` | `ESTOP` |
| `from` / `to` | 前后状态（1=触发/接管） | `0` / `1` |
| `state_duration_s` | 上一状态持续秒数 | `12.417` |
| `voltage_v` / `soc_pct` / `gear` | 事件时刻的上下文 | `51.72` / `77` / `3` |

| 参数 | 默认 | 说明 |
|---|---|---|
| `log_dir` | `~/yhs_labs/S2-05` | 日志目录（自动创建） |
| `max_bytes` | 10485760 | 单文件上限 10 MB |
| `keep_files` | 3 | 保留 3 个历史文件 |
| `chassis_topic` | `/chassis_info_fb` | 底盘反馈话题 |

## 10. 回滚与常见问题

回滚：①`Ctrl+C` 退出，日志关闭并打印汇总；②`mv ~/yhs_labs/S2-05/safety_events.csv ~/yhs_labs/S2-05/archive_$(date +%m%d).csv` 归档或直接 `rm -rf ~/yhs_labs/S2-05`；③`git status` 应无新增；④本实验不改任何仓库文件与参数，急停逻辑未被触碰。

| 现象 | 原因 | 处理 |
|---|---|---|
| 有状态但日志无记录 | 边沿检测只在变化时记录；初始态不写 CSV | 初始只打印，属预期；触发一次即可看到行 |
| `ModuleNotFoundError: No module named 'rclpy'` | 未 source ROS 2 | `source /opt/ros/humble/setup.bash`；用 `/usr/bin/python3` |
| `No module named 'yhs_can_interfaces'` | 未 source 工作空间 | `source install/setup.bash`；`ros2 pkg prefix` 验证 |
| 按急停无任何变化 | 急停回路未接入 IO 反馈，或读到旧帧 | `ros2 topic echo --field io_fb.io_fb_estop` 复核；查急停线束 |
| CSV 被 Excel 占用无法写入 | 文件被其他程序锁定（在 Windows 上编辑） | 关闭表格软件或复制副本再打开 |
| 时间与现场差 8 h | 本机时区不是 Asia/Shanghai | `timedatectl set-timezone Asia/Shanghai` 或记录时注明时区 |
| 日志行数翻倍 | 同时运行了两个留痕脚本 | `ros2 node list` 查重，只保留一个 |
