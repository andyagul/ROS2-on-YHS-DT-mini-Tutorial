## 2. 实验目标

1. 能读懂 `nav2_msgs/srv/LoadMap`（`string map_url` → `nav_msgs/OccupancyGrid map` + `uint8 result`）与 `SaveMap`（`map_topic`、`map_url`、`image_format`、`map_mode`、`free_thresh`、`occupied_thresh` → `bool result`），并据此设计封装服务。
2. 能建立带校验和的索引 YAML，加载前校验 YAML/图像存在性与 sha256，坏图 100% 拦截。
3. 能实现 `list_maps`/`switch_map`/`rollback_map`/`save_map`/`verify_maps` 五个服务，单次切换 ≤ 3 min。
4. 能保留历史版本并一键回滚，误切率 0。
5. 连续 20 次切换无异常，返回的 `map.info` 与索引场地一致。

### 6.3 src/map_manager_node.cpp（完整）

```cpp
#include <algorithm>
#include <chrono>
#include <cstdio>
#include <cstdlib>
#include <ctime>
#include <filesystem>
#include <fstream>
#include <memory>
#include <sstream>
#include <string>
#include <utility>
#include <vector>
#include "nav2_msgs/srv/load_map.hpp"
#include "nav2_msgs/srv/save_map.hpp"
#include "rclcpp/rclcpp.hpp"
#include "std_srvs/srv/trigger.hpp"
#include "yaml-cpp/yaml.h"

namespace fs = std::filesystem;
using LoadMap = nav2_msgs::srv::LoadMap;
using SaveMap = nav2_msgs::srv::SaveMap;
using Trigger = std_srvs::srv::Trigger;
using R = LoadMap::Response;
struct Entry {std::string id, site, yaml, sha256, updated_at; double speed_limit{0.3}; int version{1};};

static std::string expandUser(const std::string & p)
{
  const char * h = std::getenv("HOME");
  return p.rfind("~/", 0) == 0 ? std::string(h ? h : "") + p.substr(1) : p;
}
static std::string sha256Of(const std::string & path)
{
  FILE * f = popen(("sha256sum -- \"" + path + "\" 2>/dev/null").c_str(), "r");
  if (!f) {return "";}
  char buf[256] = {0};
  const bool ok = fgets(buf, sizeof(buf), f) != nullptr;
  pclose(f);
  const std::string s(buf);
  return ok ? s.substr(0, s.find(' ')) : "";
}
static std::string nowIso()
{
  char buf[32] = {0};
  std::time_t t = std::time(nullptr);
  std::strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%S", std::localtime(&t));
  return buf;
}

class MapManager : public rclcpp::Node
{
public:
  MapManager() : Node("map_manager_node")
  {
    index_file_ = expandUser(declare_parameter<std::string>("index_file", "~/yhs_maps/index.yaml"));
    history_dir_ = expandUser(declare_parameter<std::string>("history_dir", "~/yhs_maps/history"));
    load_service_ = declare_parameter<std::string>("load_map_service", "/map_server/load_map");
    save_service_ = declare_parameter<std::string>("save_map_service", "/map_saver/save_map");
    verify_ = declare_parameter<bool>("verify_checksum", true);
    timeout_s_ = declare_parameter<double>("service_timeout_s", 15.0);
    auto srv = create_callback_group(rclcpp::CallbackGroupType::Reentrant);
    auto cli = create_callback_group(rclcpp::CallbackGroupType::Reentrant);
    load_cli_ = create_client<LoadMap>(load_service_, rmw_qos_profile_services_default, cli);
    save_cli_ = create_client<SaveMap>(save_service_, rmw_qos_profile_services_default, cli);
    create_service<Trigger>("~/list_maps", [this](auto, auto, auto r) {onList(r);},
      rmw_qos_profile_services_default, srv);
    create_service<LoadMap>("~/switch_map", [this](auto, auto q, auto r) {onSwitch(q, r);},
      rmw_qos_profile_services_default, srv);
    create_service<LoadMap>("~/rollback_map", [this](auto, auto q, auto r) {onRollback(q, r);},
      rmw_qos_profile_services_default, srv);
    create_service<SaveMap>("~/save_map", [this](auto, auto q, auto r) {onSave(q, r);},
      rmw_qos_profile_services_default, srv);
    create_service<Trigger>("~/verify_maps", [this](auto, auto, auto r) {onVerify(r);},
      rmw_qos_profile_services_default, srv);
    RCLCPP_INFO(get_logger(), "地图管理就绪 index=%s", index_file_.c_str());
  }

private:
  void loadIndex()
  {
    entries_.clear();
    if (!fs::exists(index_file_)) {RCLCPP_WARN(get_logger(), "索引不存在: %s", index_file_.c_str()); return;}
    try {
      for (const auto & m : YAML::LoadFile(index_file_)["maps"]) {
        Entry e;
        e.id = m["id"].as<std::string>();
        e.site = m["site"] ? m["site"].as<std::string>() : "-";
        e.speed_limit = m["speed_limit"] ? m["speed_limit"].as<double>() : 0.3;
        e.version = m["version"] ? m["version"].as<int>() : 1;
        e.yaml = m["yaml"].as<std::string>();
        e.sha256 = m["sha256"] ? m["sha256"].as<std::string>() : "";
        e.updated_at = m["updated_at"] ? m["updated_at"].as<std::string>() : "";
        entries_.push_back(e);
      }
    } catch (const std::exception & e) {
      RCLCPP_ERROR(get_logger(), "索引解析失败: %s", e.what());
    }
    std::sort(entries_.begin(), entries_.end(),
      [](const Entry & a, const Entry & b) {return a.id != b.id ? a.id < b.id : a.version < b.version;});
  }
  void saveIndex() const
  {
    fs::create_directories(fs::path(index_file_).parent_path());
    std::ofstream f(index_file_);
    f << "maps:\n";
    for (const auto & e : entries_) {
      f << "  - {id: " << e.id << ", site: \"" << e.site << "\", speed_limit: " << e.speed_limit
        << ", version: " << e.version << ", yaml: \"" << e.yaml << "\", sha256: \"" << e.sha256
        << "\", updated_at: \"" << e.updated_at << "\"}\n";
    }
  }
  Entry * find(const std::string & id, int version = -1)
  {
    Entry * hit = nullptr;
    for (auto & e : entries_) {
      if (e.id != id) {continue;}
      if ((version < 0 || e.version == version) && (!hit || e.version > hit->version)) {hit = &e;}
    }
    return hit;
  }
  static std::pair<std::string, int> target(const std::string & s)
  {
    const auto at = s.find('@');
    if (at == std::string::npos) {return {s, -1};}
    return {s.substr(0, at), std::stoi(s.substr(at + 1))};
  }
  static fs::path imageOf(const std::string & yaml_path)
  {
    try {
      const std::string img = YAML::LoadFile(yaml_path)["image"].as<std::string>();
      return fs::path(img).is_absolute() ? fs::path(img) : fs::path(yaml_path).parent_path() / img;
    } catch (const std::exception &) {return {};}
  }
  uint8_t validate(const Entry & e, std::string & why)
  {
    if (!fs::exists(e.yaml)) {why = "YAML 缺失: " + e.yaml; return R::RESULT_MAP_DOES_NOT_EXIST;}
    const fs::path img = imageOf(e.yaml);
    if (img.empty() || !fs::exists(img)) {why = "图像缺失: " + img.string(); return R::RESULT_INVALID_MAP_DATA;}
    if (!verify_) {why = "已跳过校验"; return R::RESULT_SUCCESS;}
    if (e.sha256.empty()) {why = "索引未登记校验和，先调用 verify_maps"; return R::RESULT_INVALID_MAP_DATA;}
    if (sha256Of(img.string()) != e.sha256) {why = "校验和不匹配: " + img.string(); return R::RESULT_INVALID_MAP_DATA;}
    why = "校验通过";
    return R::RESULT_SUCCESS;
  }
  uint8_t doLoad(const std::string & id, int version, nav_msgs::msg::OccupancyGrid & out, std::string & why)
  {
    Entry * e = find(id, version);
    if (!e) {why = "索引中无 " + id; return R::RESULT_MAP_DOES_NOT_EXIST;}
    const uint8_t rc = validate(*e, why);
    if (rc != R::RESULT_SUCCESS) {return rc;}
    auto req = std::make_shared<LoadMap::Request>();
    req->map_url = e->yaml;  // 必须绝对路径：本仓库 map_io 不剥离 file:// 前缀
    auto fut = load_cli_->async_send_request(req);
    if (fut.wait_for(std::chrono::duration<double>(timeout_s_)) != std::future_status::ready) {
      why = "调用 " + load_service_ + " 超时";
      return R::RESULT_UNDEFINED_FAILURE;
    }
    auto resp = fut.get();
    out = resp->map;
    why = "map_server result=" + std::to_string(resp->result);
    return resp->result;
  }
  void onList(const std::shared_ptr<Trigger::Response> & r)
  {
    loadIndex();
    std::ostringstream ss;
    ss << "共 " << entries_.size() << " 条\n";
    for (const auto & e : entries_) {
      ss << "  " << e.id << " | " << e.site << " | v" << e.version << " | 限速 " << e.speed_limit
         << " | " << (e.sha256.empty() ? "未校验" : e.sha256.substr(0, 12)) << " | " << e.yaml << "\n";
    }
    r->success = !entries_.empty();
    r->message = ss.str();
  }
  void onSwitch(const std::shared_ptr<LoadMap::Request> & q, const std::shared_ptr<LoadMap::Response> & r)
  {
    loadIndex();
    const auto t = target(q->map_url);
    std::string why;
    r->result = doLoad(t.first, t.second, r->map, why);
    RCLCPP_INFO(get_logger(), "switch_map %s: %s", t.first.c_str(), why.c_str());
  }
  void onRollback(const std::shared_ptr<LoadMap::Request> & q, const std::shared_ptr<LoadMap::Response> & r)
  {
    loadIndex();
    const auto t = target(q->map_url);
    Entry * e = find(t.first);
    int version = t.second;
    if (version < 0) {version = (e && e->version > 1) ? e->version - 1 : -1;}
    if (version < 0) {r->result = R::RESULT_MAP_DOES_NOT_EXIST; RCLCPP_ERROR(get_logger(), "无可回滚版本"); return;}
    std::string why;
    r->result = doLoad(t.first, version, r->map, why);
    RCLCPP_INFO(get_logger(), "rollback %s v%d: %s", t.first.c_str(), version, why.c_str());
  }
  void onSave(const std::shared_ptr<SaveMap::Request> & q, const std::shared_ptr<SaveMap::Response> & r)
  {
    loadIndex();
    const auto t = target(q->map_url);
    Entry * e = find(t.first);
    if (!e) {RCLCPP_ERROR(get_logger(), "索引无 %s，请先登记场地", t.first.c_str()); r->result = false; return;}
    const int version = t.second > 0 ? t.second : e->version + 1;
    const fs::path dir = fs::path(history_dir_) / e->id / ("v" + std::to_string(version));
    fs::create_directories(dir);
    const std::string prefix = (dir / e->id).string();
    auto req = std::make_shared<SaveMap::Request>();
    req->map_topic = q->map_topic.empty() ? "map" : q->map_topic;
    req->map_url = prefix;
    req->image_format = q->image_format.empty() ? "pgm" : q->image_format;
    req->map_mode = q->map_mode.empty() ? "trinary" : q->map_mode;
    req->free_thresh = q->free_thresh;
    req->occupied_thresh = q->occupied_thresh;
    auto fut = save_cli_->async_send_request(req);
    if (fut.wait_for(std::chrono::duration<double>(timeout_s_)) != std::future_status::ready) {
      RCLCPP_ERROR(get_logger(), "调用 %s 超时", save_service_.c_str());
      r->result = false;
      return;
    }
    r->result = fut.get()->result;
    if (!r->result) {RCLCPP_ERROR(get_logger(), "保存失败"); return;}
    Entry n;
    n.id = e->id; n.site = e->site; n.speed_limit = e->speed_limit; n.version = version;
    n.yaml = prefix + ".yaml";
    n.sha256 = sha256Of(prefix + "." + req->image_format);
    n.updated_at = nowIso();
    entries_.push_back(n);
    saveIndex();
    RCLCPP_INFO(get_logger(), "已保存 %s v%d → %s", n.id.c_str(), version, n.yaml.c_str());
  }
  void onVerify(const std::shared_ptr<Trigger::Response> & r)
  {
    loadIndex();
    int registered = 0, bad = 0;
    std::ostringstream ss;
    for (auto & e : entries_) {
      const fs::path img = imageOf(e.yaml);
      if (img.empty() || !fs::exists(img)) {ss << e.id << " 图像缺失; "; ++bad; continue;}
      const std::string sum = sha256Of(img.string());
      if (e.sha256.empty()) {e.sha256 = sum; e.updated_at = nowIso(); ++registered;}
      else if (e.sha256 != sum) {ss << e.id << " 校验和不匹配; "; ++bad;}
    }
    saveIndex();
    r->success = bad == 0;
    r->message = "已登记 " + std::to_string(registered) + " 条，异常 " + std::to_string(bad) + " 条 " + ss.str();
  }

  std::string index_file_, history_dir_, load_service_, save_service_;
  double timeout_s_{15.0};
  bool verify_{true};
  std::vector<Entry> entries_;
  rclcpp::Client<LoadMap>::SharedPtr load_cli_;
  rclcpp::Client<SaveMap>::SharedPtr save_cli_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  rclcpp::executors::MultiThreadedExecutor exec(rclcpp::ExecutorOptions(), 2);
  exec.add_node(std::make_shared<MapManager>());
  exec.spin();
  rclcpp::shutdown();
  return 0;
}
```

## 10. 回滚与常见问题

| 现象 / 报错 | 原因 | 处理 |
|---|---|---|
| `switch_map` 返回 `result=1` | 索引路径不存在或未登记校验和 | 核对 `yaml` 绝对路径；先 `verify_maps` |
| `switch_map` 返回 `result=2` | 图像缺失、校验和不符或坏图 | 看日志；`zgq-g.yaml` 缺 `zgq.pgm` 属预期拦截 |
| `result=255` + `调用 /map_server/load_map 超时` | `map_server` 未激活或服务名不同 | `ros2 lifecycle get /map_server` 应为 `active`；`ros2 service list | grep load_map` |
| 传 `file:///...` 报 `RESULT_MAP_DOES_NOT_EXIST` | 本仓库 `map_io` 不解析 URI 前缀 | 一律传绝对路径（本包已处理） |
| `save_map` 返回 false | `map_saver` 未启动或 `/map` 无数据 | `ros2 launch nav2_map_server map_saver_server.launch.py`；确认 `/map` 有 latched 数据 |
| 切换后 RViz2 地图没变 | 只改参数或客户端缓存 | `ros2 topic echo /map --once --field info` 核对；RViz2 重新订阅 Map |
| 中文场地名乱码 | YAML 非 UTF-8 | `file ~/yhs_maps/index.yaml`，用 UTF-8 保存 |
| 回滚 | 索引或历史版本损坏 | 恢复 `~/yhs_maps/index.yaml.bak`；历史版本在 `~/yhs_maps/history/<id>/v<ver>/` |
