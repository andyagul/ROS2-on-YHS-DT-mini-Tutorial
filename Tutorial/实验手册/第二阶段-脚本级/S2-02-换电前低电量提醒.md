## 2. 实验目标

1. 能读出 `bms_fb.bms_fb_voltage`、`bms_fb.bms_fb_current`、`bms_fb.bms_fb_remaining_capacity` 与 `bms_flag_fb.bms_flag_fb_soc`，并说明单位。
2. 能设置"提醒/强制"两级阈值（SOC 与电压并联），并解释双条件的必要性。
3. 能编写 rclpy 节点，越限后 3 s 内发布 `IoCmd.io_cmd_speaker=true` 驱动喇叭响 1 s，重复提醒间隔 ≥ 60 s；并把告警写成含时间戳、电压、电流、容量、SOC、动作的 CSV 行，用于交班记录。
4. 能用命令行改阈值复现两级告警与恢复，并说明 8 h 运行误报 ≤ 1 次/班的判定方法。

## 5. 实施步骤

### 5.1 写入完整脚本

把下面全文写入 `~/yhs_labs/S2-02/battery_alert.py`（唯一副本）：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""S2-02 换电前低电量提醒：越限蜂鸣 1 s 并写 CSV 日志。"""

import os
from datetime import datetime

import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from yhs_can_interfaces.msg import ChassisInfoFb, IoCmd

RED, YEL, RST = '\033[31m', '\033[33m', '\033[0m'
HEADER = ['time', 'level', 'voltage_v', 'current_a', 'capacity_ah', 'soc_pct', 'action']


class BatteryAlert(Node):
    """两级低电量阈值 → 喇叭提示 + 交班日志。"""

    def __init__(self):
        super().__init__('battery_alert_s2_02')
        params = (('warn_soc', 25), ('crit_soc', 15), ('warn_voltage', 47.0),
                  ('crit_voltage', 45.0), ('repeat_interval', 60.0), ('beep_seconds', 1.0),
                  ('confirm_ticks', 3), ('log_dir', '~/yhs_labs/S2-02'))
        for name, value in params:
            self.declare_parameter(name, value)
        self.warn_soc = int(self.get_parameter('warn_soc').value)
        self.crit_soc = int(self.get_parameter('crit_soc').value)
        self.warn_voltage = float(self.get_parameter('warn_voltage').value)
        self.crit_voltage = float(self.get_parameter('crit_voltage').value)
        self.repeat_interval = float(self.get_parameter('repeat_interval').value)
        self.beep_seconds = float(self.get_parameter('beep_seconds').value)
        self.confirm_ticks = int(self.get_parameter('confirm_ticks').value)
        log_dir = os.path.expanduser(str(self.get_parameter('log_dir').value))
        os.makedirs(log_dir, exist_ok=True)
        self.log_path = os.path.join(log_dir, 'battery_alert.log')
        self.log_file = open(self.log_path, 'a', buffering=1)
        if os.path.getsize(self.log_path) == 0:
            self.log_file.write(','.join(HEADER) + '\n')
        qos = QoSProfile(depth=10, reliability=ReliabilityPolicy.RELIABLE,
                         history=HistoryPolicy.KEEP_LAST)
        self.sub = self.create_subscription(ChassisInfoFb, '/chassis_info_fb', self.on_chassis, qos)
        self.pub = self.create_publisher(IoCmd, '/io_cmd', 10)
        self.msg = None
        self.level = 'OK'
        self.candidate, self.candidate_ticks = 'OK', 0
        self.last_alert, self.beep_until = 0.0, 0.0
        self.warn_count, self.crit_count = 0, 0
        self.create_timer(1.0, self.evaluate)
        self.create_timer(0.1, self.publish_io)
        self.get_logger().info(f'低电量提醒已启动：SOC {self.warn_soc}/{self.crit_soc} %，'
                               f'电压 {self.warn_voltage}/{self.crit_voltage} V，日志 {self.log_path}')
    def now_s(self):
        return self.get_clock().now().nanoseconds / 1e9
    def on_chassis(self, msg):
        self.msg = msg
    def evaluate(self):
        if self.msg is None:
            return
        v = self.msg.bms_fb.bms_fb_voltage
        soc = int(self.msg.bms_flag_fb.bms_flag_fb_soc)
        if v <= 0.0:
            return
        if v <= self.crit_voltage or soc <= self.crit_soc:
            level = 'CRIT'
        elif v <= self.warn_voltage or soc <= self.warn_soc:
            level = 'WARN'
        else:
            level = 'OK'
        if level == self.level:
            self.candidate_ticks = 0
            if level != 'OK':
                self.alert(level, v, soc, repeat=True)
            return
        if level != self.candidate:
            self.candidate, self.candidate_ticks = level, 1
        else:
            self.candidate_ticks += 1
        if self.candidate_ticks < self.confirm_ticks:
            return
        old, self.level = self.level, level
        self.candidate_ticks = 0
        if level == 'OK':
            self.log_row('OK', v, soc, 'clear')
            self.get_logger().info(f'电压/SOC 回到安全区（原 {old}），提醒解除')
        else:
            self.alert(level, v, soc, repeat=False)
    def alert(self, level, v, soc, repeat):
        now = self.now_s()
        if repeat and now - self.last_alert < self.repeat_interval:
            return
        self.last_alert = now
        self.beep_until = now + self.beep_seconds
        self.log_row(level, v, soc, f'speaker_{self.beep_seconds:.1f}s')
        text = (f'电压 {v:.2f} V  SOC {soc}%  剩余 '
                f'{self.msg.bms_fb.bms_fb_remaining_capacity:.2f} Ah')
        if level == 'CRIT':
            self.crit_count += 1
            print(RED + '[强制换电] ' + text + RST, flush=True)
            self.get_logger().error('强制换电告警：' + text)
        else:
            self.warn_count += 1
            print(YEL + '[提醒换电] ' + text + RST, flush=True)
            self.get_logger().warn('提醒换电告警：' + text)
    def log_row(self, level, v, soc, action):
        row = [datetime.now().strftime('%Y-%m-%d %H:%M:%S.%f')[:-3], level, f'{v:.2f}',
               f'{self.msg.bms_fb.bms_fb_current:.2f}',
               f'{self.msg.bms_fb.bms_fb_remaining_capacity:.2f}', str(soc), action]
        self.log_file.write(','.join(row) + '\n')
    def publish_io(self):
        msg = IoCmd()
        msg.io_cmd_speaker = self.now_s() < self.beep_until
        self.pub.publish(msg)
    def shutdown(self):
        msg = IoCmd()
        for _ in range(3):
            self.pub.publish(msg)
        self.log_file.close()
        self.get_logger().info(f'告警退出：提醒 {self.warn_count} 次，强制 {self.crit_count} 次')

def main(args=None):
    rclpy.init(args=args)
    node = BatteryAlert()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.shutdown()
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()


if __name__ == '__main__':
    main()
```

## 6. 关键代码与配置

| 代码段 | 作用 | 关键点 |
|---|---|---|
| `params` + `declare_parameter` | 8 个可调参数（阈值、去抖、日志目录） | 全部可用 `-p` 覆盖，现场不改代码 |
| 日志初始化 | 建目录、追加打开、空文件写表头 | `buffering=1` 行缓冲，断电不丢整段 |
| `QoSProfile` + 订阅 | 订阅 `/chassis_info_fb` | 与底盘 publisher 的 RELIABLE/KEEP_LAST(1) 兼容 |
| `evaluate`（1 s） | 算等级 + 连续 3 拍确认 + 升降级 | 去抖，避免电压瞬时跌落误报 |
| `alert` | 置 `beep_until`、写日志、分级打印 | 同级重复受 `repeat_interval`（默认 60 s）限制 |
| `publish_io`（10 Hz） | `io_cmd_speaker = now < beep_until` | 驱动中 `sendDataTemp[2] = msg.io_cmd_speaker`，不依赖灯控使能位 |

| 话题 | 类型 | 方向 | 说明 |
|---|---|---|---|
| `/chassis_info_fb` | `yhs_can_interfaces/msg/ChassisInfoFb` | 订阅 | RELIABLE，发布端 depth 1 |
| `/io_cmd` | `yhs_can_interfaces/msg/IoCmd` | 发布 | depth 10，10 Hz 周期发送 |

字段与阈值（48 V 包示例，按实车标称值调整）：

| 字段 | 单位 | 提醒（WARN） | 强制（CRIT） |
|---|---|---|---|
| `bms_flag_fb.bms_flag_fb_soc` | % | ≤ 25 | ≤ 15 |
| `bms_fb.bms_fb_voltage` | V | ≤ 47.0 | ≤ 45.0 |
| `bms_fb.bms_fb_remaining_capacity` | Ah | 仅记录 | 仅记录 |
| `bms_fb.bms_fb_current` | A | 仅记录 | 仅记录 |

> 判定为"或"逻辑：SOC 与电压任一越限即报警。24 V 平台改为 `-p warn_voltage:=23.5 -p crit_voltage:=22.5`。

## 10. 回滚与常见问题

回滚：①`Ctrl+C` 退出（脚本先发 3 次 `speaker=false`）；②确认喇叭已停，`/io_cmd` 最后一条 `io_cmd_speaker` 为 `false`；③清理数据 `rm -rf ~/yhs_labs/S2-02`；④`git status` 应无新增；⑤参数均走命令行，无文件需还原。

| 现象 | 原因 | 处理 |
|---|---|---|
| `ModuleNotFoundError: No module named 'rclpy'` | 未 source ROS 2 或用非系统 Python | `source /opt/ros/humble/setup.bash`；用 `/usr/bin/python3` |
| `No module named 'yhs_can_interfaces'` | 未 source 工作空间或未编译 | `source install/setup.bash`；`ros2 pkg prefix yhs_can_interfaces` |
| 不报警 | 阈值仍低于当前值，或电压为 0 被跳过 | 用 `-p warn_soc:=90 -p crit_soc:=85` 复现；先修 BMS 数据 |
| 喇叭不响但 `/io_cmd` 有 true | 喇叭硬件/供电异常，或底盘未收到 | `candump can0` 查 `0x98C4D7D0` 报文；查喇叭线束 |
| 喇叭长鸣 | 脚本被 `kill -9`，未执行 `shutdown` | 重跑脚本并 `Ctrl+C`，或重启底盘节点 |
