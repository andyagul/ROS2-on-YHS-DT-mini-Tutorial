# S3-01 车队健康管理与预防性维护

> 场景来源：[二次开发场景实施方案.md(../二次开发场景实施方案.md) S3-01；清单条目：[二次开发项目清单.md(../二次开发项目清单.md) 第三阶段 #1。
> 本实验新建 C++ 功能包 `yhs_chassis_monitor`：订阅 `/chassis_info_fb`，聚合轮速/IO/BMS，用 `diagnostic_updater` 向 `/diagnostics` 发布 5 类诊断项与健康评分。全程只读，不参与控制。下文 `WS="$HOME/ros2_ws"`。

## 2. 实验目标

1. 能按 ROS 2 规范建 `ament_cmake` 包，并说明 `package.xml` 中 `rclcpp`、`diagnostic_updater`、`diagnostic_msgs`、`yhs_can_interfaces` 四条依赖各被哪段代码使用。
2. 能说明 `ChassisInfoFb` 六个子结构（`ctrl_fb`/`io_fb`/`l_wheel_fb`/`r_wheel_fb`/`bms_flag_fb`/`bms_fb`）分别来自哪帧 CAN 反馈，并正确取用 SOC、电压、6 路碰撞、6 路跌落、急停字段。
3. 能让 `/diagnostics` 稳定输出 5 类诊断项（通信链路、电池 BMS、轮速驱动、安全 IO、健康评分），`name`/`level`/`values` 可被 `rqt_runtime_monitor` 直接消费。
4. 能用 YAML 把阈值分级为 OK/WARN/ERROR，并计算 0~100 健康评分；参数热更新后下一周期（≤ 1 s）生效。
5. 能注入"低压、断链、碰撞"三类故障，验证等级与评分随之变化，且监控节点异常不影响底盘。

### 6.3 `src/chassis_monitor_node.cpp`

```cpp
#include <algorithm>
#include <chrono>
#include <cmath>
#include <memory>
#include <mutex>
#include <string>
#include <utility>
#include <vector>

#include "diagnostic_updater/diagnostic_updater.hpp"
#include "rclcpp/rclcpp.hpp"
#include "yhs_can_interfaces/msg/chassis_info_fb.hpp"

using ChassisInfoFb = yhs_can_interfaces::msg::ChassisInfoFb;
using diagnostic_msgs::msg::DiagnosticStatus;
using Clock = std::chrono::steady_clock;

class ChassisMonitorNode : public rclcpp::Node
{
public:
  ChassisMonitorNode() : Node("yhs_chassis_monitor"), updater_(this, 1.0)
  {
    const std::vector<std::pair<std::string, double>> def{
      {"voltage_warn", 24.0}, {"voltage_error", 23.0}, {"soc_warn", 30.0}, {"soc_error", 20.0},
      {"link_warn_timeout", 1.0}, {"link_error_timeout", 3.0},
      {"wheel_speed_max", 1.5}, {"wheel_diff_warn", 0.2}, {"temp_warn", 55.0}};
    for (const auto & kv : def) {declare_parameter(kv.first, kv.second);}

    sub_ = create_subscription<ChassisInfoFb>(
      "/chassis_info_fb", rclcpp::QoS(rclcpp::KeepLast(1)).reliable(),
      [this](const ChassisInfoFb::SharedPtr m) {
        std::lock_guard<std::mutex> lock(mutex_);
        data_ = *m; last_rx_ = Clock::now(); has_data_ = true; ++frames_;
      });

    updater_.setHardwareID("YHS_DT_mini");
    updater_.add("Chassis/通信链路", [this](diagnostic_updater::DiagnosticStatusWrapper & s) {diag_link(s);});
    updater_.add("Chassis/电池BMS", [this](diagnostic_updater::DiagnosticStatusWrapper & s) {diag_battery(s);});
    updater_.add("Chassis/轮速驱动", [this](diagnostic_updater::DiagnosticStatusWrapper & s) {diag_wheel(s);});
    updater_.add("Chassis/安全IO", [this](diagnostic_updater::DiagnosticStatusWrapper & s) {diag_io(s);});
    updater_.add("Chassis/健康评分", [this](diagnostic_updater::DiagnosticStatusWrapper & s) {diag_health(s);});
    RCLCPP_INFO(get_logger(), "健康监控已启动：订阅 /chassis_info_fb，发布 /diagnostics（周期见 diagnostic_updater.period）");
  }

private:
  double p(const std::string & n) {return get_parameter(n).as_double();}
  double age() const {return std::chrono::duration<double>(Clock::now() - last_rx_).count();}
  static int cnt6(bool a, bool b, bool c, bool d, bool e, bool f)
  {return int(a) + int(b) + int(c) + int(d) + int(e) + int(f);}
  int impacts() const
  {const auto & i = data_.io_fb; return cnt6(i.io_fb_fl_impact_sensor, i.io_fb_fm_impact_sensor, i.io_fb_fr_impact_sensor, i.io_fb_rl_impact_sensor, i.io_fb_rm_impact_sensor, i.io_fb_rr_impact_sensor);}
  int drops() const
  {const auto & i = data_.io_fb; return cnt6(i.io_fb_fl_drop_sensor, i.io_fb_fm_drop_sensor, i.io_fb_fr_drop_sensor, i.io_fb_rl_drop_sensor, i.io_fb_rm_drop_sensor, i.io_fb_rr_drop_sensor);}

  void diag_link(diagnostic_updater::DiagnosticStatusWrapper & s)
  {
    std::lock_guard<std::mutex> lock(mutex_);
    s.add("topic", "/chassis_info_fb"); s.add("frames", frames_);
    if (!has_data_) {
      s.summary(DiagnosticStatus::ERROR, "从未收到底盘反馈，检查 can0 与 yhs_can_control_node");
      return;
    }
    const double a = age(); s.addf("age_s", "%.3f", a);
    if (a > p("link_error_timeout")) {
      s.summaryf(DiagnosticStatus::ERROR, "底盘反馈中断 %.2f s（阈值 %.1f s）", a, p("link_error_timeout"));
    } else if (a > p("link_warn_timeout")) {
      s.summaryf(DiagnosticStatus::WARN, "底盘反馈延迟 %.2f s（阈值 %.1f s）", a, p("link_warn_timeout"));
    } else {
      s.summaryf(DiagnosticStatus::OK, "链路正常，累计 %lu 帧", static_cast<unsigned long>(frames_));
    }
  }

  void diag_battery(diagnostic_updater::DiagnosticStatusWrapper & s)
  {
    std::lock_guard<std::mutex> lock(mutex_);
    const auto & b = data_.bms_fb; const auto & f = data_.bms_flag_fb; const int soc = f.bms_flag_fb_soc;
    s.addf("voltage_V", "%.2f", b.bms_fb_voltage); s.addf("current_A", "%.2f", b.bms_fb_current);
    s.add("soc_percent", soc);
    s.addf("temp_C", "high=%.1f low=%.1f", f.bms_flag_fb_hight_temperature, f.bms_flag_fb_low_temperature);
    s.add("flag_uv", f.bms_flag_fb_uv); s.add("flag_short", f.bms_flag_fb_short);
    if (!has_data_) {s.summary(DiagnosticStatus::STALE, "未收到 BMS 数据"); return;}
    std::string err, warn;
    if (b.bms_fb_voltage < p("voltage_error")) err += " 电压低于强制下线阈值";
    if (soc < p("soc_error")) err += " SOC 低于强制阈值";
    if (f.bms_flag_fb_uv || f.bms_flag_fb_short || f.bms_flag_fb_ic_error) err += " BMS 告警位置位";
    if (b.bms_fb_voltage < p("voltage_warn")) warn += " 电压偏低";
    if (soc < p("soc_warn")) warn += " SOC 偏低";
    if (f.bms_flag_fb_hight_temperature > p("temp_warn")) warn += " 电池温度偏高";
    if (!err.empty()) {
      s.summaryf(DiagnosticStatus::ERROR, "电压 %.2f V / SOC %d%%：%s", b.bms_fb_voltage, soc, err.c_str());
    } else if (!warn.empty()) {
      s.summaryf(DiagnosticStatus::WARN, "电压 %.2f V / SOC %d%%：%s", b.bms_fb_voltage, soc, warn.c_str());
    } else {
      s.summaryf(DiagnosticStatus::OK, "电压 %.2f V / SOC %d%% / 最高温 %.1f ℃",
        b.bms_fb_voltage, soc, f.bms_flag_fb_hight_temperature);
    }
  }

  void diag_wheel(diagnostic_updater::DiagnosticStatusWrapper & s)
  {
    std::lock_guard<std::mutex> lock(mutex_);
    const double l = data_.l_wheel_fb.l_wheel_fb_velocity;
    const double r = data_.r_wheel_fb.r_wheel_fb_velocity;
    s.addf("wheel_left_mps", "%.3f", l); s.addf("wheel_right_mps", "%.3f", r);
    s.addf("ctrl_linear_mps", "%.3f", data_.ctrl_fb.ctrl_fb_linear);
    s.addf("wheel_diff_mps", "%.3f", std::fabs(l - r));
    if (!has_data_) {s.summary(DiagnosticStatus::STALE, "无轮速数据"); return;}
    const double mx = std::max(std::fabs(l), std::fabs(r));
    if (mx > p("wheel_speed_max")) {
      s.summaryf(DiagnosticStatus::ERROR, "轮速超限 %.2f m/s（阈值 %.2f）", mx, p("wheel_speed_max"));
    } else if (std::fabs(l - r) > p("wheel_diff_warn") && mx > 0.05) {
      s.summaryf(DiagnosticStatus::WARN, "左右轮速差 %.2f m/s，疑似打滑或单侧堵转", std::fabs(l - r));
    } else {
      s.summaryf(DiagnosticStatus::OK, "左右轮速 %.2f / %.2f m/s 一致", l, r);
    }
  }

  void diag_io(diagnostic_updater::DiagnosticStatusWrapper & s)
  {
    std::lock_guard<std::mutex> lock(mutex_);
    const auto & io = data_.io_fb;
    const int imp = impacts(); const int drp = drops();
    s.add("impact_count", imp); s.add("drop_count", drp);
    s.add("estop", io.io_fb_estop); s.add("joypad_ctrl", io.io_fb_joypad_ctrl);
    if (!has_data_) {s.summary(DiagnosticStatus::STALE, "无 IO 反馈"); return;}
    std::string err;
    if (imp > 0) err += " 碰撞(" + std::to_string(imp) + "路)";
    if (drp > 0) err += " 跌落(" + std::to_string(drp) + "路)";
    if (io.io_fb_estop) err += " 急停按下";
    if (!err.empty()) {
      s.summaryf(DiagnosticStatus::ERROR, "安全信号触发：%s", err.c_str());
    } else if (io.io_fb_joypad_ctrl) {
      s.summary(DiagnosticStatus::WARN, "手柄接管中，底盘由人工控制");
    } else {
      s.summary(DiagnosticStatus::OK, "碰撞/跌落/急停均未触发");
    }
  }

  void diag_health(diagnostic_updater::DiagnosticStatusWrapper & s)
  {
    std::lock_guard<std::mutex> lock(mutex_);
    int score = 100; std::string detail;
    const int soc = data_.bms_flag_fb.bms_flag_fb_soc;
    if (!has_data_) {
      score = 0; detail = " 无底盘数据";
    } else {
      if (age() > p("link_error_timeout")) {score -= 40; detail += " 链路中断";}
      else if (age() > p("link_warn_timeout")) {score -= 15; detail += " 链路延迟";}
      if (data_.bms_fb.bms_fb_voltage < p("voltage_error")) {score -= 30; detail += " 电压过低";}
      else if (data_.bms_fb.bms_fb_voltage < p("voltage_warn")) {score -= 10; detail += " 电压偏低";}
      if (soc < p("soc_error")) {score -= 25; detail += " SOC 过低";}
      else if (soc < p("soc_warn")) {score -= 8; detail += " SOC 偏低";}
      if (impacts() > 0) {score -= 40; detail += " 碰撞";}
      if (drops() > 0) {score -= 40; detail += " 跌落";}
      if (data_.io_fb.io_fb_estop) {score -= 30; detail += " 急停";}
      if (data_.io_fb.io_fb_joypad_ctrl) {score -= 5; detail += " 手柄接管";}
    }
    score = std::max(0, score);
    const char * grade = score >= 80 ? "优" : (score >= 60 ? "良" : (score >= 40 ? "关注" : "故障"));
    s.add("score", score); s.add("grade", grade);
    s.add("reasons", detail.empty() ? "无" : detail);
    if (score >= 80) s.summaryf(DiagnosticStatus::OK, "健康评分 %d（%s）", score, grade);
    else if (score >= 60) s.summaryf(DiagnosticStatus::WARN, "健康评分 %d（%s）：%s", score, grade, detail.c_str());
    else s.summaryf(DiagnosticStatus::ERROR, "健康评分 %d（%s）：%s", score, grade, detail.c_str());
  }

  diagnostic_updater::Updater updater_;
  rclcpp::Subscription<ChassisInfoFb>::SharedPtr sub_;
  std::mutex mutex_;
  ChassisInfoFb data_;
  Clock::time_point last_rx_{Clock::now()};
  bool has_data_{false};
  unsigned long frames_{0UL};
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<ChassisMonitorNode>());
  rclcpp::shutdown();
  return 0;
}
```

### 6.4 `config/thresholds.yaml`

```yaml
yhs_chassis_monitor:
  ros__parameters:
    diagnostic_updater.period: 1.0   # /diagnostics 周期（s），由 Updater 自己声明
    voltage_warn: 24.0               # 24 V 系统低于此值告 WARN
    voltage_error: 23.0              # 低于此值告 ERROR，必须回库换电
    soc_warn: 30.0
    soc_error: 20.0
    link_warn_timeout: 1.0           # 超过 1 s 无新帧 → WARN
    link_error_timeout: 3.0          # 超过 3 s → ERROR
    wheel_speed_max: 1.5             # 轮速绝对上限（m/s）
    wheel_diff_warn: 0.2             # 左右轮速差阈值（m/s）
    temp_warn: 55.0                  # 电池最高温告警（℃）
```

```python
import os

from ament_index_python.packages import get_package_share_directory
from launch import LaunchDescription
from launch.actions import DeclareLaunchArgument
from launch.substitutions import LaunchConfiguration
from launch_ros.actions import Node


def generate_launch_description():
    share = get_package_share_directory('yhs_chassis_monitor')
    params_file = LaunchConfiguration('params_file')
    chassis_topic = LaunchConfiguration('chassis_topic')

    return LaunchDescription([
        DeclareLaunchArgument(
            'params_file',
            default_value=os.path.join(share, 'config', 'thresholds.yaml'),
            description='阈值与发布周期参数文件'),
        DeclareLaunchArgument(
            'chassis_topic', default_value='/chassis_info_fb',
            description='底盘反馈话题，多车时改 /car2/chassis_info_fb'),
        Node(
            package='yhs_chassis_monitor',
            executable='yhs_chassis_monitor_node',
            name='yhs_chassis_monitor',
            output='screen',
            parameters=[params_file],
            remappings=[('/chassis_info_fb', chassis_topic)],
        ),
    ])
```

|---|---|---|

## 10. 回滚与常见问题

```bash
# 1) A 终端 Ctrl+C 停止监控节点
rm -rf "$WS"/build/yhs_chassis_monitor "$WS"/install/yhs_chassis_monitor   # 2) 删编译安装产物
rm -rf "$WS"/src/yhs_chassis_monitor                                      # 3) 删源码包
source "$WS"/install/setup.bash && ros2 pkg list | grep yhs_chassis_monitor  # 4) 应无输出
git -C "$WS" status --short                                                # 5) 应无残留改动
```

| 现象 | 原因 | 处理 |
|---|---|---|
| 找不到 `diagnostic_updater` 的 CMake 配置 | 未装诊断库 | `sudo apt install ros-humble-diagnostic-updater` |
| `chassis_info_fb.hpp: No such file` | 未 source 工作空间或接口包未编 | `colcon build --packages-select yhs_can_interfaces` 后重新 source |
| 编译报 `Updater` 无该成员 | Humble 头文件是 `diagnostic_updater/diagnostic_updater.hpp` | 按 6.3 的 include 书写 |
| `/diagnostics` 无输出 | 节点未起，或 `diagnostic_updater.period` 被设过大 | `ros2 node list`、`ros2 param get /yhs_chassis_monitor diagnostic_updater.period` |
| 只出现 1~2 个诊断项 | 构造函数中途抛异常或回调崩溃 | 看 A 终端 `terminate called`；逐个 `add` 回加定位 |
| 报 `parameter 'diagnostic_updater.period' has already been declared` | 同一节点重复构造 `Updater` | `Updater` 必须是节点成员且只构造一次 |
| `age_s` 持续增大 | `can0` 未 up、`can_name` 不符、底盘未上电 | `ip -details link show can0`、`candump can0` |
| 诊断值恒为 0 且 `从未收到` | 订阅 QoS 不兼容或话题被重映射错 | `ros2 topic info /chassis_info_fb -v` 核对 QoS 与 launch 的 `chassis_topic` |
