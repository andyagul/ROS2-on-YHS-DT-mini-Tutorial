## 2. 实验目标

1. 能指出真实跟随接口：action `navigate_follow_record_path`、类型 `yhs_can_interfaces/action/RecordPath`、话题入口 `/record_path`（`nav_msgs/Path`），并说明二者关系。
2. 能录制 `/amcl_pose` 生成 `nav_msgs/Path`，实现抽稀、平滑（滑动平均 + 重算航向）、追加拼接。
3. 能设计可版本管理的路线 YAML（点位、限速、点数、长度），加载耗时 ≤ 2 s。
4. 能实现"偏离路径 > 0.3 m 取消导航"，并给出横向误差量化测量方法与记录表。
5. 完成 20 次复现，横向误差最大值 ≤ 0.2 m、成功率 ≥ 95%，输出改线作业流程。

### 6.3 src/path_teach_node.cpp（完整）

```cpp
#include <algorithm>
#include <cmath>
#include <cstdlib>
#include <filesystem>
#include <fstream>
#include <iomanip>
#include <memory>
#include <string>
#include <vector>
#include "geometry_msgs/msg/pose_with_covariance_stamped.hpp"
#include "nav_msgs/msg/path.hpp"
#include "rclcpp/rclcpp.hpp"
#include "rclcpp_action/rclcpp_action.hpp"
#include "std_srvs/srv/trigger.hpp"
#include "yaml-cpp/yaml.h"
#include "yhs_can_interfaces/action/record_path.hpp"

namespace fs = std::filesystem;
using Trigger = std_srvs::srv::Trigger;
using RecordPath = yhs_can_interfaces::action::RecordPath;
struct Pt {double x{0.0}; double y{0.0}; double yaw{0.0};};

class PathTeach : public rclcpp::Node
{
public:
  PathTeach() : Node("path_teach_node")
  {
    route_dir_ = expandUser(declare_parameter<std::string>("route_dir", "~/yhs_routes"));
    route_name_ = declare_parameter<std::string>("route_name", "route_a");
    amcl_topic_ = declare_parameter<std::string>("amcl_topic", "/amcl_pose");
    record_path_topic_ = declare_parameter<std::string>("record_path_topic", "/record_path");
    min_dist_ = declare_parameter<double>("min_record_dist", 0.03);
    decim_min_ = declare_parameter<double>("decimate_min_dist", 0.10);
    smooth_window_ = declare_parameter<int>("smooth_window", 5);
    speed_limit_ = declare_parameter<double>("speed_limit", 0.30);
    max_allowed_error_ = declare_parameter<double>("max_allowed_error", 0.30);
    append_ = declare_parameter<bool>("append_on_start", false);
    path_pub_ = create_publisher<nav_msgs::msg::Path>(record_path_topic_, rclcpp::QoS(10));
    act_ = rclcpp_action::create_client<RecordPath>(this, "navigate_follow_record_path");
    sub_ = create_subscription<geometry_msgs::msg::PoseWithCovarianceStamped>(amcl_topic_,
      rclcpp::QoS(20),
      [this](const geometry_msgs::msg::PoseWithCovarianceStamped::SharedPtr m) {onPose(m);});
    create_service<Trigger>("~/start_record", [this](auto, auto, auto r) {onStart(r);});
    create_service<Trigger>("~/stop_record", [this](auto, auto, auto r) {onStop(r);});
    create_service<Trigger>("~/load_path", [this](auto, auto, auto r) {onLoad(r);});
    create_service<Trigger>("~/publish_path", [this](auto, auto, auto r) {onPublish(r);});
    create_service<Trigger>("~/cancel_follow", [this](auto, auto, auto r) {onCancel(r);});
    create_service<Trigger>("~/stats", [this](auto, auto, auto r) {onStats(r);});
    RCLCPP_INFO(get_logger(), "示教就绪 %s route=%s", route_dir_.c_str(), route_name_.c_str());
  }

private:
  void refresh() {
    route_name_ = get_parameter("route_name").as_string();
    append_ = get_parameter("append_on_start").as_bool();
  }
  std::string routeFile() const {return route_dir_ + "/" + route_name_ + ".yaml";}
  double length() const {
    double s = 0.0;
    for (size_t i = 1; i < pts_.size(); ++i) {s += std::hypot(pts_[i].x - pts_[i - 1].x, pts_[i].y - pts_[i - 1].y);}
    return s;
  }
  double lateralError(double x, double y) const {
    double best = 1e9;
    for (size_t i = 1; i < pts_.size(); ++i) {
      const double ax = pts_[i - 1].x, ay = pts_[i - 1].y;
      const double dx = pts_[i].x - ax, dy = pts_[i].y - ay, l2 = dx * dx + dy * dy;
      double t = l2 > 1e-12 ? ((x - ax) * dx + (y - ay) * dy) / l2 : 0.0;
      t = std::clamp(t, 0.0, 1.0);
      best = std::min(best, std::hypot(x - (ax + t * dx), y - (ay + t * dy)));
    }
    return best;
  }
  void onPose(const geometry_msgs::msg::PoseWithCovarianceStamped::SharedPtr m) {
    const auto & p = m->pose.pose;
    const double yaw = std::atan2(2.0 * p.orientation.w * p.orientation.z,
      1.0 - 2.0 * p.orientation.z * p.orientation.z);
    if (evaluating_ && pts_.size() >= 2) {
      const double err = lateralError(p.position.x, p.position.y);
      max_error_ = std::max(max_error_, err);
      if (err > max_allowed_error_ && !stopped_ && act_->action_server_is_ready()) {
        stopped_ = true;
        act_->async_cancel_all_goals();
        RCLCPP_ERROR(get_logger(), "横向误差 %.3f m 超限 %.2f m，取消导航", err, max_allowed_error_);
      }
    }
    if (!recording_) {return;}
    if (!pts_.empty() && std::hypot(p.position.x - pts_.back().x,
      p.position.y - pts_.back().y) < min_dist_) {return;}
    pts_.push_back({p.position.x, p.position.y, yaw});
    if (pts_.size() % 100 == 0) {
      RCLCPP_INFO(get_logger(), "已录 %zu 点 %.1f m", pts_.size(), length());
    }
  }
  void onStart(const std::shared_ptr<Trigger::Response> & r) {
    refresh();
    if (recording_) {r->success = false; r->message = "已在录制中"; return;}
    if (!append_) {pts_.clear();}
    recording_ = true; evaluating_ = false; stopped_ = false;
    r->success = true; r->message = append_ ? "追加录制" : "开始录制";
  }
  void onStop(const std::shared_ptr<Trigger::Response> & r) {
    refresh(); recording_ = false;
    if (pts_.size() < 2) {r->success = false; r->message = "录制点不足"; return;}
    const size_t raw = pts_.size();
    pts_ = smooth(decimate(pts_));
    if (pts_.size() < 2) {r->success = false; r->message = "处理后点数不足"; return;}
    r->success = save();
    r->message = (r->success ? "已保存 " : "保存失败 ") + routeFile() + "：" + std::to_string(raw) +
      " → " + std::to_string(pts_.size()) + " 点，" + std::to_string(length()) + " m";
  }
  void onLoad(const std::shared_ptr<Trigger::Response> & r) {
    refresh(); r->success = load();
    r->message = r->success ? "已加载 " + routeFile() : "加载失败 " + routeFile();
  }
  void onPublish(const std::shared_ptr<Trigger::Response> & r) {
    refresh();
    if (pts_.size() < 2) {r->success = false; r->message = "无路径"; return;}
    nav_msgs::msg::Path path;
    path.header.frame_id = "map"; path.header.stamp = now();
    for (const auto & p : pts_) {
      geometry_msgs::msg::PoseStamped ps;
      ps.header = path.header;
      ps.pose.position.x = p.x; ps.pose.position.y = p.y;
      ps.pose.orientation.z = std::sin(p.yaw / 2.0); ps.pose.orientation.w = std::cos(p.yaw / 2.0);
      path.poses.push_back(ps);
    }
    path_pub_->publish(path);
    evaluating_ = true; stopped_ = false; max_error_ = 0.0;
    r->success = true;
    r->message = "已发布 " + std::to_string(path.poses.size()) + " 点至 " + record_path_topic_;
  }
  void onCancel(const std::shared_ptr<Trigger::Response> & r) {
    evaluating_ = false; r->success = act_->action_server_is_ready();
    if (r->success) {act_->async_cancel_all_goals();}
    r->message = r->success ? "已发送取消" : "action 不可用";
  }
  void onStats(const std::shared_ptr<Trigger::Response> & r) {
    r->success = true;
    r->message = "points=" + std::to_string(pts_.size()) + " max_error=" + std::to_string(max_error_);
  }
  std::vector<Pt> decimate(const std::vector<Pt> & in) const {
    if (in.size() < 3 || decim_min_ <= 0.0) {return in;}
    std::vector<Pt> out{in.front()};
    for (size_t i = 1; i + 1 < in.size(); ++i) {
      if (std::hypot(in[i].x - out.back().x, in[i].y - out.back().y) >= decim_min_) {out.push_back(in[i]);}
    }
    out.push_back(in.back());
    return out;
  }
  std::vector<Pt> smooth(const std::vector<Pt> & in) const {
    std::vector<Pt> out = in;
    const int half = smooth_window_ / 2;
    if (in.size() >= 5 && half >= 1) {
      for (size_t i = 0; i < in.size(); ++i) {
        double sx = 0.0, sy = 0.0;
        int n = 0;
        for (int k = -half; k <= half; ++k) {
          const int j = static_cast<int>(i) + k;
          if (j < 0 || j >= static_cast<int>(in.size())) {continue;}
          sx += in[j].x; sy += in[j].y; ++n;
        }
        out[i].x = sx / n; out[i].y = sy / n;
      }
    }
    for (size_t i = 0; i < out.size(); ++i) {
      const size_t a = (i == 0) ? 0 : i - 1, b = (i + 1 < out.size()) ? i + 1 : i;
      if (b > a) {out[i].yaw = std::atan2(out[b].y - out[a].y, out[b].x - out[a].x);}
    }
    return out;
  }
  bool save() const {
    fs::create_directories(fs::path(routeFile()).parent_path());
    std::ofstream f(routeFile());
    if (!f) {return false;}
    f << std::fixed << std::setprecision(4) << "route:\n  name: " << route_name_
      << "\n  frame_id: map\n  point_count: " << pts_.size() << "\n  length_m: " << length()
      << "\n  speed_limit: " << speed_limit_ << "\n  poses:\n";
    for (const auto & p : pts_) {f << "    - {x: " << p.x << ", y: " << p.y << ", yaw: " << p.yaw << "}\n";}
    return true;
  }
  bool load() {
    if (!fs::exists(routeFile())) {return false;}
    try {
      std::vector<Pt> v;
      for (const auto & n : YAML::LoadFile(routeFile())["route"]["poses"]) {
        v.push_back({n["x"].as<double>(), n["y"].as<double>(), n["yaw"].as<double>()});
      }
      if (v.size() < 2) {return false;}
      pts_ = v;
      return true;
    } catch (const std::exception & e) {
      RCLCPP_ERROR(get_logger(), "解析失败: %s", e.what());
      return false;
    }
  }
  static std::string expandUser(const std::string & p) {
    const char * home = std::getenv("HOME");
    return p.rfind("~/", 0) == 0 ? std::string(home ? home : "") + p.substr(1) : p;
  }

  std::string route_dir_, route_name_, amcl_topic_, record_path_topic_;
  double min_dist_{0.03}, decim_min_{0.10}, speed_limit_{0.30}, max_allowed_error_{0.30}, max_error_{0.0};
  int smooth_window_{5};
  bool append_{false}, recording_{false}, evaluating_{false}, stopped_{false};
  std::vector<Pt> pts_;
  rclcpp::Publisher<nav_msgs::msg::Path>::SharedPtr path_pub_;
  rclcpp::Subscription<geometry_msgs::msg::PoseWithCovarianceStamped>::SharedPtr sub_;
  rclcpp_action::Client<RecordPath>::SharedPtr act_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<PathTeach>());
  rclcpp::shutdown();
  return 0;
}
```

判定：处理后点数 < 原始点数；长度与皮尺/里程计偏差 ≤ 2%；`max_error` ≤ 0.2 m；取消后停车 ≤ 0.5 s。超限保护演练：`ros2 param set /path_teach_node max_allowed_error 0.05` 后复现，应打印 `横向误差 ... 超限 0.05 m` 并自动取消（仅架空/低速，演练后恢复 0.30）。

横向误差测量方法：①在线（主）——`publish_path` 后进入评估态，`stats` 返回本次全程最大误差，每次复现后取一次；②离线复核——`ros2 bag record /amcl_pose` 后按同一公式复算（位姿 P 与路径相邻点 A、B，`t=clip(((P-A)·(B-A))/|B-A|²,0,1)`，误差 `=|P-(A+t(B-A))|`，全程取最大）；③口径——取"全程最大值"，同时记录均值。

## 10. 回滚与常见问题

| 现象 / 报错 | 原因 | 处理 |
|---|---|---|
| `cancel_follow` 返回 `action 不可用` | bt_navigator 未激活 | `ros2 lifecycle get /bt_navigator` 应为 `active`，确认 `autostart:=true` |
| 发布后无 `Received record path` | 话题不匹配 | `ros2 topic info /record_path -v` 看订阅者；`use_namespace:=true` 时话题为 `/<ns>/record_path` |
| `BT file not found: . Navigation canceled.` | 手工填了非法 `behavior_tree` | 本包不填该字段，留空即用默认跟随 BT |
| 路径被 TEB 拉直、贴墙 | 点位过密或含定位跳变 | `decimate_min_dist` 提到 0.15~0.2、`smooth_window` 提到 7；避开 AMCL 收敛前路段 |
| `max_error` 恒为 0 | 未调用 `publish_path` | 先发布路径使节点进入评估态 |
| 回滚 | 本实验不改上游配置 | 删除/重编 `yhs_path_teach` 不影响 `yhs_nav2`；路线回滚恢复 `~/yhs_routes/*.yaml` 备份 |
