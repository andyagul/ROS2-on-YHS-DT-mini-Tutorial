## 2. 实验目标

1. 能指出可用的真实事件源字段（`io_fb.io_fb_estop`、`io_fb.io_fb_joypad_ctrl`、`io_fb.io_fb_{fl,fm,fr,rl,rm,rr}_impact_sensor`、`io_fb.io_fb_*_drop_sensor`、`bms_flag_fb.bms_flag_fb_soc`）并做边沿检测。
2. 能说明 rosbag2 子进程与 C++ API 两种录制方式的取舍，并给出选择依据。
3. 能实现事件触发录制：事件时刻、事件前 30 s、事件后 15 s 的数据可完整复现，单次切片 ≤ 200 MB。
4. 能导出 CSV 时间线（位姿、速度、状态、事件），按 t=0 对齐事件时刻。
5. 能实现掉电/异常退出后切片仍可用，且磁盘不足时自动停录。
6. 完成 3 类事件演练（急停、碰撞、人工接管），切片可用率 100%。

### 6.3 src/blackbox_node.cpp（完整）

```cpp
#include <sys/stat.h>
#include <sys/statvfs.h>
#include <sys/wait.h>
#include <unistd.h>
#include <algorithm>
#include <chrono>
#include <csignal>
#include <cstdlib>
#include <ctime>
#include <filesystem>
#include <fstream>
#include <iomanip>
#include <memory>
#include <sstream>
#include <string>
#include <thread>
#include <vector>
#include "geometry_msgs/msg/pose_with_covariance_stamped.hpp"
#include "nav_msgs/msg/odometry.hpp"
#include "rclcpp/rclcpp.hpp"
#include "std_srvs/srv/trigger.hpp"
#include "yhs_can_interfaces/msg/chassis_info_fb.hpp"

namespace fs = std::filesystem;
using Trigger = std_srvs::srv::Trigger;
using Chassis = yhs_can_interfaces::msg::ChassisInfoFb;

static std::string expandUser(const std::string & p)
{
  const char * h = std::getenv("HOME");
  return p.rfind("~/", 0) == 0 ? std::string(h ? h : "") + p.substr(1) : p;
}
static std::string stamp()
{
  char buf[32] = {0};
  std::time_t t = std::time(nullptr);
  std::strftime(buf, sizeof(buf), "%Y%m%d_%H%M%S", std::localtime(&t));
  return buf;
}
static std::string iso()
{
  char buf[32] = {0};
  std::time_t t = std::time(nullptr);
  std::strftime(buf, sizeof(buf), "%Y-%m-%dT%H:%M:%S", std::localtime(&t));
  return buf;
}

class Blackbox : public rclcpp::Node
{
public:
  Blackbox() : Node("blackbox_node")
  {
    root_ = expandUser(declare_parameter<std::string>("bag_root", "~/yhs_blackbox"));
    topics_ = declare_parameter<std::vector<std::string>>("topics",
      {"/chassis_info_fb", "/odom", "/amcl_pose", "/cmd_vel"});
    seg_s_ = declare_parameter<int>("segment_duration", 60);
    size_mb_ = declare_parameter<int>("max_seg_size_mb", 100);
    pre_s_ = declare_parameter<int>("pre_event_s", 30);
    post_s_ = declare_parameter<int>("post_event_s", 15);
    keep_h_ = declare_parameter<int>("retention_hours", 72);
    min_gb_ = declare_parameter<double>("min_free_gb", 5.0);
    fs::create_directories(segDir());
    fs::create_directories(evDir());
    create_service<Trigger>("~/trigger", [this](auto, auto, auto r) {onManual(r);});
    create_service<Trigger>("~/status", [this](auto, auto, auto r) {onStatus(r);});
    ch_sub_ = create_subscription<Chassis>("/chassis_info_fb", rclcpp::QoS(20),
      [this](const Chassis::SharedPtr m) {onChassis(m);});
    odom_sub_ = create_subscription<nav_msgs::msg::Odometry>("/odom", rclcpp::QoS(20),
      [this](const nav_msgs::msg::Odometry::SharedPtr m) {onOdom(m);});
    pose_sub_ = create_subscription<geometry_msgs::msg::PoseWithCovarianceStamped>("/amcl_pose",
      rclcpp::QoS(20), [this](const geometry_msgs::msg::PoseWithCovarianceStamped::SharedPtr m) {
        x_ = m->pose.pose.position.x; y_ = m->pose.pose.position.y;
      });
    timer_ = create_wall_timer(std::chrono::seconds(1), [this]() {tick();});
    if (!startRecorder()) {RCLCPP_ERROR(get_logger(), "rosbag2 启动失败");}
    RCLCPP_INFO(get_logger(), "黑匣子就绪 root=%s 分段=%ds", root_.c_str(), seg_s_);
  }
  ~Blackbox() override {stopRecorder();}

private:
  fs::path segDir() const {return fs::path(root_) / "segments";}
  fs::path evDir() const {return fs::path(root_) / "events";}
  bool startRecorder()
  {
    if (pid_ > 0) {return true;}
    std::vector<std::string> args{"ros2", "bag", "record", "-o",
      (segDir() / ("seg_" + stamp())).string(),
      "--max-bag-duration", std::to_string(seg_s_),
      "--max-bag-size", std::to_string(size_mb_ * 1024 * 1024)};
    for (const auto & t : topics_) {args.push_back(t);}
    pid_ = fork();
    if (pid_ == 0) {
      setpgid(0, 0);
      std::vector<char *> av;
      for (auto & a : args) {av.push_back(const_cast<char *>(a.c_str()));}
      av.push_back(nullptr);
      execvp("ros2", av.data());
      _exit(127);
    }
    return pid_ > 0;
  }
  void stopRecorder()
  {
    if (pid_ <= 0) {return;}
    kill(pid_, SIGINT);
    for (int i = 0; i < 60 && waitpid(pid_, nullptr, WNOHANG) == 0; ++i) {
      std::this_thread::sleep_for(std::chrono::milliseconds(100));
    }
    kill(pid_, SIGKILL);
    waitpid(pid_, nullptr, 0);
    pid_ = -1;
  }
  void onChassis(const Chassis::SharedPtr m)
  {
    estop_ = m->io_fb.io_fb_estop;
    joy_ = m->io_fb.io_fb_joypad_ctrl;
    impact_ = m->io_fb.io_fb_fl_impact_sensor || m->io_fb.io_fb_fm_impact_sensor ||
      m->io_fb.io_fb_fr_impact_sensor || m->io_fb.io_fb_rl_impact_sensor ||
      m->io_fb.io_fb_rm_impact_sensor || m->io_fb.io_fb_rr_impact_sensor;
    drop_ = m->io_fb.io_fb_fl_drop_sensor || m->io_fb.io_fb_fm_drop_sensor ||
      m->io_fb.io_fb_fr_drop_sensor || m->io_fb.io_fb_rl_drop_sensor ||
      m->io_fb.io_fb_rm_drop_sensor || m->io_fb.io_fb_rr_drop_sensor;
    soc_ = m->bms_flag_fb.bms_flag_fb_soc;
    lin_ = m->ctrl_fb.ctrl_fb_linear;
    ang_ = m->ctrl_fb.ctrl_fb_angular;
    if (estop_ && !estop_prev_) {raise("ESTOP", "io_fb_estop");}
    if (impact_ && !impact_prev_) {raise("IMPACT", "impact_sensor");}
    if (drop_ && !drop_prev_) {raise("DROP", "drop_sensor");}
    if (joy_ && !joy_prev_) {raise("TAKEOVER", "joypad_ctrl");}
    estop_prev_ = estop_; impact_prev_ = impact_; drop_prev_ = drop_; joy_prev_ = joy_;
  }
  void onOdom(const nav_msgs::msg::Odometry::SharedPtr m)
  {
    vx_ = m->twist.twist.linear.x;
    wz_ = m->twist.twist.angular.z;
  }
  void raise(const std::string & type, const std::string & src)
  {
    Slice s;
    s.type = type; s.src = src; s.dir = evDir() / ("EVT_" + stamp() + "_" + type);
    s.t = std::time(nullptr);
    s.deadline = s.t + post_s_;
    s.estop = estop_; s.impact = impact_; s.drop = drop_;
    s.soc = soc_; s.lin = lin_; s.ang = ang_; s.x = x_; s.y = y_;
    s.copy_done = false;
    fs::create_directories(s.dir);
    appendCsv(s);
    pending_.push_back(s);
    RCLCPP_WARN(get_logger(), "事件 %s (%s) → %s", type.c_str(), src.c_str(), s.dir.c_str());
  }
  void appendCsv(const Slice & s)
  {
    const fs::path p = fs::path(root_) / "events.csv";
    const bool fresh = !fs::exists(p);
    std::ofstream f(p, std::ios::app);
    if (fresh) {
      f << "time_iso,event,source,estop,impact,drop,soc,linear,angular,x,y,slice\n";
    }
    f << iso() << "," << s.type << "," << s.src << "," << s.estop << "," << s.impact
      << "," << s.drop << "," << int(s.soc) << "," << s.lin << "," << s.ang << "," << s.x
      << "," << s.y << "," << s.dir.string() << "\n";
  }
  void tick()
  {
    const std::time_t now = std::time(nullptr);
    const bool due = std::any_of(pending_.begin(), pending_.end(),
      [now](const Slice & s) {return !s.copy_done && now >= s.deadline;});
    if (due) {
      const bool was_recording = pid_ > 0;
      if (was_recording) {stopRecorder();}   // 封卷，保证切片是完整 sqlite
      for (auto & s : pending_) {
        if (!s.copy_done && now >= s.deadline) {
          copySegments(s);
          s.copy_done = true;
        }
      }
      if (was_recording) {startRecorder();}
    }
    pending_.erase(std::remove_if(pending_.begin(), pending_.end(),
      [](const Slice & s) {return s.copy_done;}), pending_.end());
    const double free_gb = freeGb();
    if (free_gb < min_gb_ && pid_ > 0) {
      stopRecorder();
      RCLCPP_ERROR(get_logger(), "磁盘剩余 %.1f GB 低于阈值，已停录", free_gb);
    }
    cleanup();
  }
  void copySegments(Slice & s)
  {
    uintmax_t bytes = 0;
    int copied = 0;
    for (const auto & d : fs::directory_iterator(segDir())) {
      if (!d.is_directory()) {continue;}
      struct stat st {};
      if (stat(d.path().c_str(), &st) != 0) {continue;}
      const std::time_t mt = st.st_mtime;
      if (mt < s.t - pre_s_ - seg_s_ || mt > s.deadline + seg_s_) {continue;}
      std::error_code ec;
      fs::copy(d.path(), s.dir / d.path().filename(), fs::copy_options::recursive, ec);
      if (ec) {continue;}
      bytes += sizeOf(d.path());
      if (++copied >= 2 || bytes >= 200ull * 1024 * 1024) {break;}
    }
    std::ofstream meta(s.dir / "meta.yaml");
    meta << "event: " << s.type << "\nsource: " << s.src << "\ntime: " << iso()
         << "\npre_s: " << pre_s_ << "\npost_s: " << post_s_ << "\nsegments: " << copied
         << "\nbytes: " << bytes << "\nframe: map\nx: " << s.x << "\ny: " << s.y
         << "\nestop: " << s.estop << "\nimpact: " << s.impact << "\ndrop: " << s.drop
         << "\nsoc: " << int(s.soc) << "\nlinear: " << s.lin << "\nangular: " << s.ang << "\n";
  }
  static uintmax_t sizeOf(const fs::path & p)
  {
    uintmax_t n = 0;
    for (const auto & e : fs::recursive_directory_iterator(p)) {
      if (e.is_regular_file()) {n += e.file_size();}
    }
    return n;
  }
  double freeGb() const
  {
    struct statvfs vfs {};
    if (statvfs(root_.c_str(), &vfs) != 0) {return 0.0;}
    return double(vfs.f_bavail) * double(vfs.f_frsize) / (1024.0 * 1024.0 * 1024.0);
  }
  void cleanup()
  {
    const std::time_t now = std::time(nullptr);
    for (const auto & d : fs::directory_iterator(segDir())) {
      if (!d.is_directory()) {continue;}
      struct stat st {};
      if (stat(d.path().c_str(), &st) != 0) {continue;}
      if (now - st.st_mtime > keep_h_ * 3600) {
        std::error_code ec;
        fs::remove_all(d.path(), ec);
      }
    }
  }
  void onManual(const std::shared_ptr<Trigger::Response> & r)
  {
    raise("MANUAL", "service");
    r->success = true;
    r->message = pending_.empty() ? "事件已记录" : ("已生成事件切片 " + pending_.back().dir.string());
  }
  void onStatus(const std::shared_ptr<Trigger::Response> & r)
  {
    int segs = 0;
    for (const auto & d : fs::directory_iterator(segDir())) {
      if (d.is_directory()) {++segs;}
    }
    std::ostringstream ss;
    ss << "recording=" << (pid_ > 0 ? 1 : 0) << " pending=" << pending_.size()
       << " free_gb=" << std::fixed << std::setprecision(1) << freeGb()
       << " segments=" << segs << " estop=" << estop_ << " soc=" << int(soc_);
    r->success = true;
    r->message = ss.str();
  }

  struct Slice {
    std::string type, src;
    fs::path dir;
    std::time_t t{0}, deadline{0};
    bool estop{false}, impact{false}, drop{false}, copy_done{false};
    uint8_t soc{0};
    double lin{0}, ang{0}, x{0}, y{0};
  };
  std::string root_;
  std::vector<std::string> topics_;
  int seg_s_{60}, size_mb_{100}, pre_s_{30}, post_s_{15}, keep_h_{72};
  double min_gb_{5.0};
  bool estop_{false}, joy_{false}, impact_{false}, drop_{false};
  bool estop_prev_{false}, joy_prev_{false}, impact_prev_{false}, drop_prev_{false};
  uint8_t soc_{0};
  double lin_{0}, ang_{0}, vx_{0}, wz_{0}, x_{0}, y_{0};
  int pid_{-1};
  std::vector<Slice> pending_;
  rclcpp::Subscription<Chassis>::SharedPtr ch_sub_;
  rclcpp::Subscription<nav_msgs::msg::Odometry>::SharedPtr odom_sub_;
  rclcpp::Subscription<geometry_msgs::msg::PoseWithCovarianceStamped>::SharedPtr pose_sub_;
  rclcpp::TimerBase::SharedPtr timer_;
};

int main(int argc, char ** argv)
{
  rclcpp::init(argc, argv);
  rclcpp::spin(std::make_shared<Blackbox>());
  rclcpp::shutdown();
  return 0;
}
```

### 6.6 scripts/export_timeline.py（完整）

```python
#!/usr/bin/env python3
"""把一个事件切片里的 bag 导出为 4 个 CSV，t=0 对齐事件时刻。"""
import csv
import sys
from pathlib import Path

import rosbag2_py
from rclpy.serialization import deserialize_message
from rosidl_runtime_py.utilities import get_message


def reader(path):
    r = rosbag2_py.SequentialReader()
    r.open(rosbag2_py.StorageOptions(uri=str(path), storage_id='sqlite3'),
           rosbag2_py.ConverterOptions('', ''))
    types = {t.name: t.type for t in r.get_all_topics_and_types()}
    return r, types


def main(slice_dir):
    slice_dir = Path(slice_dir)
    bags = sorted(slice_dir.glob('seg_*'))
    if not bags:
        sys.exit('未找到 seg_* 目录')
    out = {'pose': [], 'odom': [], 'chassis': []}
    t0 = None
    for bag in bags:
        r, types = reader(bag)
        while r.has_next():
            topic, data, t = r.read_next()
            stamp = t / 1e9
            if t0 is None or stamp < t0:
                t0 = stamp
            msg = deserialize_message(data, get_message(types[topic]))
            if topic == '/amcl_pose':
                p = msg.pose.pose
                out['pose'].append([stamp, p.position.x, p.position.y, p.orientation.z, p.orientation.w])
            elif topic == '/odom':
                tw = msg.twist.twist
                out['odom'].append([stamp, tw.linear.x, tw.angular.z])
            elif topic == '/chassis_info_fb':
                io, cf, bf = msg.io_fb, msg.ctrl_fb, msg.bms_flag_fb
                impact = any([io.io_fb_fl_impact_sensor, io.io_fb_fm_impact_sensor,
                              io.io_fb_fr_impact_sensor, io.io_fb_rl_impact_sensor,
                              io.io_fb_rm_impact_sensor, io.io_fb_rr_impact_sensor])
                drop = any([io.io_fb_fl_drop_sensor, io.io_fb_fm_drop_sensor,
                            io.io_fb_fr_drop_sensor, io.io_fb_rl_drop_sensor,
                            io.io_fb_rm_drop_sensor, io.io_fb_rr_drop_sensor])
                out['chassis'].append([stamp, int(io.io_fb_estop), int(io.io_fb_joypad_ctrl),
                                       int(impact), int(drop), int(bf.bms_flag_fb_soc),
                                       cf.ctrl_fb_linear, cf.ctrl_fb_angular])
    header = {
        'pose': ['t', 'x', 'y', 'qz', 'qw'],
        'odom': ['t', 'vx', 'wz'],
        'chassis': ['t', 'estop', 'joypad', 'impact', 'drop', 'soc', 'linear', 'angular'],
    }
    for name, rows in out.items():
        with open(slice_dir / f'{name}.csv', 'w', newline='') as f:
            w = csv.writer(f)
            w.writerow(header[name])
            for row in rows:
                w.writerow([f'{row[0] - t0:.3f}'] + row[1:])
    print(f'导出 {slice_dir}：pose={len(out["pose"])} odom={len(out["odom"])} '
          f'chassis={len(out["chassis"])}，t0={t0:.3f}')


if __name__ == '__main__':
    main(sys.argv[1] if len(sys.argv) > 1 else '.')
```

### 6.7 录制方式取舍（子进程 rosbag2 vs C++ API）

| 方案 | 实现代价 | 优点 | 缺点 | 本项目结论 |
|---|---|---|---|---|
| `ros2 bag record` 子进程 | 约 40 行 `fork/execvp`，无编译期依赖 | 与 CLI 行为一致、支持 `--max-bag-duration`/`--max-bag-size` 分段、升级 rosbag2 不用改代码 | 需自行管理 PID/信号、切片前要封卷（本实现事件+15 s 时 `SIGINT` 停录→复制→重启） | **采用**：教学可解释、跨版本稳定 |
| `rosbag2_cpp::Writer` / `rosbag2_transport::Recorder` C++ API | 约 120 行，需链接 rosbag2 各库 | 进程内零 IPC、可精确定制消息过滤与内存环形缓冲 | API 在 Foxy→Humble 间多次变更、分段/清理逻辑要自己写、与参数与 QoS 耦合深 | 不采用，仅在需要"事件前 30 s 零落盘"的纯内存环形缓冲时考虑 |

## 10. 回滚与常见问题

| 现象 / 报错 | 原因 | 处理 |
|---|---|---|
| 节点起不来并报 `rosbag2 启动失败` | `ros2` 不在 PATH 或 `rosbag2_transport` 未装 | `which ros2`；`sudo apt install ros-humble-rosbag2`，在 sourced 环境启动 |
| `events/EVT_*` 目录为空 | 事件后未等到 `post_event_s` 或分段尚未落盘 | 等待 ≥ 20 s；确认 `segments/` 有 `seg_*_0` 目录 |
| 切片缺事件前数据 | `segment_duration` 太小或刚启动即刻触发 | 保证 `segment_duration ≥ pre_event_s`，启动 60 s 后再演练 |
| `export_timeline.py` 报 `sqlite3` 打开失败 | 切片里是 mcap 存储 | 录制加 `--storage sqlite3`（默认即 sqlite3），或改用 mcap 插件 |
| 磁盘被写满 | 未按 `retention_hours` 清理 | 停录后手工 `rm -rf ~/yhs_blackbox/segments/seg_*`，并检查 `min_free_gb` |
| 事件重复触发（抖动） | 传感器信号抖动 | 已做上升沿检测；仍抖动则在 `onChassis` 增加 300 ms 去抖窗口 |
| 时间线对不齐 | 用了不同时钟源 | 统一 `/use_sim_time:=false`，`t0` 取切片内最早消息时间 |
| 回滚 | 录制影响导航性能或隐私要求提高 | 停节点并 `kill -INT` rosbag 子进程；`rm -rf ~/yhs_blackbox` 清除数据；本包不改上游配置 |
