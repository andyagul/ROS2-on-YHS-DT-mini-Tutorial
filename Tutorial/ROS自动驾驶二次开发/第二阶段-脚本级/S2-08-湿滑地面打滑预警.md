## 2. 实验目标

1. 能解释为什么比较的是"平移增量"而不是绝对位姿：`map` 与 `odom` 是两套坐标系，绝对坐标不可直接相减，而刚体变换不改变位移长度。
2. 能实现 2 s 滑动窗口 + 最近邻时间对齐（容差 0.3 s），并说明对齐误差上限 ≈ 车速 × 容差。
3. 能给出偏差定义 `dev = |Δodom - Δamcl|` 与分级阈值（注意 0.08 m、告警 0.15 m）。
4. 能处理数据不足（AMCL 静止时不更新）与时间不同步两种降级场景，不崩溃、不误报，并实测 3 次打滑告警、正常路面 1 h 误报 ≤ 1 次。

### 5.1 写入完整脚本

把下面全文写入 `~/yhs_labs/S2-08/slip_monitor.py`：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""S2-08 湿滑地面打滑预警：2 s 窗口比较 /odom 与 /amcl_pose 的平移增量。"""

import math
import sys
from collections import deque

import rclpy
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from geometry_msgs.msg import PoseWithCovarianceStamped
from nav_msgs.msg import Odometry

RED, YEL, GRN, RST = '\033[31m', '\033[33m', '\033[32m', '\033[0m'


class SlipMonitor(Node):
    """只读监控：偏差超阈打印告警，不下发任何控制指令。"""

    def __init__(self):
        super().__init__('slip_monitor_s2_08')
        params = (('odom_topic', '/odom'), ('amcl_topic', '/amcl_pose'),
                  ('window', 2.0), ('warn_dev', 0.08), ('alarm_dev', 0.15),
                  ('min_motion', 0.05), ('sync_tol', 0.3), ('print_period', 1.0),
                  ('debounce', 3.0))
        for name, value in params:
            self.declare_parameter(name, value)
        self.odom_topic = str(self.get_parameter('odom_topic').value)
        self.amcl_topic = str(self.get_parameter('amcl_topic').value)
        self.window = float(self.get_parameter('window').value)
        self.warn_dev = float(self.get_parameter('warn_dev').value)
        self.alarm_dev = float(self.get_parameter('alarm_dev').value)
        self.min_motion = float(self.get_parameter('min_motion').value)
        self.sync_tol = float(self.get_parameter('sync_tol').value)
        self.print_period = float(self.get_parameter('print_period').value)
        self.debounce = float(self.get_parameter('debounce').value)
        qos = QoSProfile(depth=10, reliability=ReliabilityPolicy.RELIABLE,
                         history=HistoryPolicy.KEEP_LAST)
        self.odom_buf = deque(maxlen=800)
        self.amcl_buf = deque(maxlen=800)
        self.create_subscription(Odometry, self.odom_topic, self.on_odom, qos)
        self.create_subscription(PoseWithCovarianceStamped, self.amcl_topic, self.on_amcl, qos)
        self.level = None
        self.last_print = -1e9
        self.last_alarm_t = -1e9
        self.windows = 0
        self.insufficient = 0
        self.alarms = 0
        self.counts = {'静止': 0, '正常': 0, '注意': 0, '打滑告警': 0}
        self.create_timer(0.2, self.evaluate)
        self.get_logger().info(f'打滑监测已启动：{self.odom_topic} vs {self.amcl_topic}，'
                               f'窗口 {self.window:.1f} s，告警阈值 {self.alarm_dev:.2f} m')

    def now_s(self):
        return self.get_clock().now().nanoseconds / 1e9

    def stamp_s(self, stamp):
        t = stamp.sec + stamp.nanosec * 1e-9
        return t if t > 0.0 else self.now_s()

    def on_odom(self, msg):
        p = msg.pose.pose.position
        self.odom_buf.append((self.stamp_s(msg.header.stamp), p.x, p.y))
        self.prune()

    def on_amcl(self, msg):
        p = msg.pose.pose.position
        self.amcl_buf.append((self.stamp_s(msg.header.stamp), p.x, p.y))
        self.prune()

    def prune(self):
        cutoff = self.now_s() - max(4.0, self.window * 3)
        for buf in (self.odom_buf, self.amcl_buf):
            while buf and buf[0][0] < cutoff:
                buf.popleft()

    def nearest(self, buf, t):
        best, best_dt = None, float('inf')
        for s in buf:
            dt = abs(s[0] - t)
            if dt < best_dt:
                best, best_dt = s, dt
        return best if best is not None and best_dt <= self.sync_tol else None

    def dist(self, a, b):
        return math.hypot(b[1] - a[1], b[2] - a[2])

    def evaluate(self):
        if len(self.odom_buf) < 2 or len(self.amcl_buf) < 2:
            return
        t_end = min(self.odom_buf[-1][0], self.amcl_buf[-1][0])
        t_start = t_end - self.window
        o1, o0 = self.nearest(self.odom_buf, t_end), self.nearest(self.odom_buf, t_start)
        a1, a0 = self.nearest(self.amcl_buf, t_end), self.nearest(self.amcl_buf, t_start)
        now = self.now_s()
        if None in (o1, o0, a1, a0):
            self.insufficient += 1      # 降级：AMCL 静止时不更新，或数据断流，不崩溃
            if now - self.last_print > 3.0:
                self.last_print = now
                self.get_logger().warn('窗口端点缺少同步样本，本轮跳过（检查 /odom、'
                                       '/amcl_pose 是否在线）')
            return
        d_odom, d_amcl = self.dist(o0, o1), self.dist(a0, a1)
        dev = abs(d_odom - d_amcl)
        ratio = d_amcl / d_odom if d_odom > 1e-6 else 0.0
        if d_odom < self.min_motion:
            level = '静止'
        elif dev >= self.alarm_dev:
            level = '打滑告警'
        elif dev >= self.warn_dev:
            level = '注意'
        else:
            level = '正常'
        self.windows += 1
        self.counts[level] += 1
        changed = level != self.level
        self.level = level
        if level == '静止':
            if changed:
                print(f'[静止] Δodom={d_odom:.3f} m，未做偏差判定', flush=True)
            return
        if level == '打滑告警' and now - self.last_alarm_t >= self.debounce:
            self.alarms += 1
            self.last_alarm_t = now
            print(RED + f'[打滑告警] 窗口 {self.window:.1f} s：Δodom={d_odom:.3f} m，'
                  f'Δamcl={d_amcl:.3f} m，偏差 {dev:.3f} m ≥ {self.alarm_dev:.2f} m，'
                  f'比值 {ratio:.2f}' + RST, flush=True)
            self.get_logger().error(f'疑似打滑：偏差 {dev:.3f} m（Δodom {d_odom:.3f} / '
                                    f'Δamcl {d_amcl:.3f}）')
        elif changed or now - self.last_print >= self.print_period:
            color = YEL if level == '注意' else GRN
            print(color + f'[{level}] Δodom={d_odom:.3f} m  Δamcl={d_amcl:.3f} m  '
                  f'偏差={dev:.3f} m  比值={ratio:.3f}' + RST, flush=True)
        if changed:
            self.get_logger().info(f'状态切换为 {level}')
        self.last_print = now

    def shutdown(self):
        self.get_logger().info(f'监测退出：{self.windows} 个窗口，样本不足 {self.insufficient} 次，'
                               f'告警 {self.alarms} 次，分级 {self.counts}')


def main(args=None):
    rclpy.init(args=args)
    node = SlipMonitor()
    try:
        rclpy.spin(node)
    except KeyboardInterrupt:
        pass
    finally:
        node.shutdown()
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()
    sys.exit(0)


if __name__ == '__main__':
    main()
```

## 6. 关键代码与配置

| 代码段 | 作用 | 关键点 |
|---|---|---|
| `stamp_s()` | 统一时间基准 | 优先用 `header.stamp`，为 0 时回退节点时钟；两者都是 ROS 时间，`use_sim_time: False` 下与墙钟一致 |
| `nearest(buf, t)` | 最近邻时间对齐 | 容差 `sync_tol=0.3 s`；0.2 m/s 下对齐误差 ≤ 0.06 m，小于最小告警阈值 |
| 三级判定 | 静止/正常/注意/打滑告警 | `d_odom < min_motion` 视为静止不判；0.08 m 提示、0.15 m 告警 |
| `debounce` | 告警去抖 3 s | 一次打滑只计一次，避免连续刷屏 |
| `insufficient` | 数据不足降级 | AMCL 静止时不更新、雷达/里程计断流时只告警不崩溃 |
| `QoSProfile(RELIABLE, KEEP_LAST(10))` | 订阅 | 与 `/amcl_pose`（RELIABLE + TRANSIENT_LOCAL）和 `/odom`（depth 1）都兼容 |

| 话题 | 类型 | 方向 | 频率 | 说明 |
|---|---|---|---|---|
| `/odom` | `nav_msgs/msg/Odometry` | 订阅 | ~50 Hz | 底盘轮速积分，打滑时会"虚增" |
| `/amcl_pose` | `geometry_msgs/msg/PoseWithCovarianceStamped` | 订阅 | 约 2~10 Hz | 激光匹配结果，打滑时几乎不前进 |

| 参数 | 默认 | 说明 |
|---|---|---|
| `window` | 2.0 | 滑动窗口（s），实施方案指定 2 s |
| `warn_dev` | 0.08 | 注意级偏差（m） |
| `alarm_dev` | 0.15 | 告警级偏差（m） |
| `min_motion` | 0.05 | 窗口内位移小于此值视为静止，不判定 |
| `sync_tol` | 0.3 | 时间对齐容差（s） |
| `debounce` | 3.0 | 告警去抖（s） |

## 10. 回滚与常见问题

回滚：①`Ctrl+C` 退出，终端打印统计汇总；②`rm -rf ~/yhs_labs/S2-08`；③`git status` 应无新增；④本实验不改 AMCL/底盘参数，无需还原；⑤清理湿滑区地面积水，恢复警戒线。

| 现象 | 原因 | 处理 |
|---|---|---|
| 一直"窗口端点缺少同步样本" | AMCL 未收敛或静止不更新；`/odom` 断流 | 先 `ros2 topic hz /amcl_pose` 确认；车辆动起来再看 |
| 偏差恒定在 0.1~0.3 m 但车没打滑 | 初始位姿错误，AMCL 正在收敛 | 重新下发 `2D Pose Estimate`，等待收敛后再测 |
| 手动推车却完全不告警 | 推车时 AMCL 也同步移动（轮子随动） | 架空车轮空转，或单侧悬空制造差速 |
| 偏差偶发尖峰 | 时间对齐容差内抖动、AMCL 跳变 | 观察连续 3 个窗口；必要时把 `sync_tol` 调小到 0.15 |
| 告警刷屏 | `debounce` 被改小 | 恢复 3.0 s |
