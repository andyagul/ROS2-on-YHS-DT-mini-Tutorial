## 2. 实验目标

1. 能核对 `RecordPath` action 的真实接口（名称、类型、goal/feedback/result 字段），不凭猜测写代码。
2. 能以 5 Hz 采样 `/amcl_pose`，按"位移 ≥ 0.10 m 或转角 ≥ 0.15 rad"抽稀，生成 `nav_msgs/Path` 并保存为可读 YAML。
3. 能从 YAML 还原 `nav_msgs/Path` 并通过 `navigate_follow_record_path` action 回放，且说得清 `/record_path` 话题是导航器订阅的另一个真实入口。
4. 能处理 goal 被拒绝、结果超时、人工取消三种情况，并给出可判定的退出码。
5. 能实测复现横向误差 ≤ 0.2 m、路径加载 ≤ 2 s，30 min 内完成"录线 → 复现"闭环。

### 5.1 写入完整脚本

把下面全文写入 `~/yhs_labs/S2-09/path_teach.py`（`mode:=record` 录制，`mode:=replay` 回放）：

```python
#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""S2-09 示教路径录制/回放：/amcl_pose → YAML → navigate_follow_record_path。"""

import math
import os
import sys
from datetime import datetime

import rclpy
import yaml
from rclpy.action import ActionClient
from rclpy.node import Node
from rclpy.qos import QoSProfile, ReliabilityPolicy, HistoryPolicy
from action_msgs.msg import GoalStatus
from geometry_msgs.msg import PoseStamped, PoseWithCovarianceStamped
from nav_msgs.msg import Path
from yhs_can_interfaces.action import RecordPath

STATUS = {GoalStatus.STATUS_SUCCEEDED: 'SUCCEEDED', GoalStatus.STATUS_CANCELED: 'CANCELED',
          GoalStatus.STATUS_ABORTED: 'ABORTED'}


def yaw_of(q):
    return math.atan2(2.0 * (q.w * q.z + q.x * q.y), 1.0 - 2.0 * (q.y * q.y + q.z * q.z))


class PathTeach(Node):
    def __init__(self):
        super().__init__('path_teach_s2_09')
        for name, value in (('mode', 'record'), ('path_file', '~/yhs_labs/S2-09/detour.yaml'),
                            ('frame_id', 'map'), ('sample_rate', 5.0), ('min_spacing', 0.10),
                            ('min_yaw', 0.15), ('result_timeout', 600.0)):
            self.declare_parameter(name, value)
        self.mode = str(self.get_parameter('mode').value)
        self.path_file = os.path.expanduser(str(self.get_parameter('path_file').value))
        self.frame_id = str(self.get_parameter('frame_id').value)
        self.rate = float(self.get_parameter('sample_rate').value)
        self.min_spacing = float(self.get_parameter('min_spacing').value)
        self.min_yaw = float(self.get_parameter('min_yaw').value)
        self.timeout = float(self.get_parameter('result_timeout').value)
        self.latest, self.points = None, []
        self.goal_handle, self.last_fb = None, 0.0
        if self.mode == 'record':
            qos = QoSProfile(depth=10, reliability=ReliabilityPolicy.RELIABLE,
                             history=HistoryPolicy.KEEP_LAST)
            self.create_subscription(PoseWithCovarianceStamped, '/amcl_pose', self.on_pose, qos)
            self.create_timer(1.0 / self.rate, self.sample)
            self.get_logger().info(f'录制开始：{self.rate:.0f} Hz，抽稀 {self.min_spacing:.2f} m')
        else:
            self.path = self.load_path()

    def on_pose(self, msg):
        self.latest = msg                  # 只缓存最新一帧，采样由定时器驱动

    def sample(self):
        if self.latest is None:
            return
        p = self.latest.pose.pose.position
        yaw = yaw_of(self.latest.pose.pose.orientation)
        if self.points:
            px, py, pyaw = self.points[-1]
            dyaw = abs(math.atan2(math.sin(yaw - pyaw), math.cos(yaw - pyaw)))
            if math.hypot(p.x - px, p.y - py) < self.min_spacing and dyaw < self.min_yaw:
                return
        self.points.append((p.x, p.y, yaw))

    def save(self):
        if len(self.points) < 2:
            self.get_logger().error('点数不足 2，未保存')
            return False
        length = sum(math.hypot(b[0] - a[0], b[1] - a[1])
                     for a, b in zip(self.points, self.points[1:]))
        doc = {'frame_id': self.frame_id, 'created': datetime.now().strftime('%F %T'),
               'sample_rate': self.rate, 'min_spacing': self.min_spacing,
               'point_count': len(self.points), 'length_m': round(length, 3),
               'poses': [{'x': round(x, 4), 'y': round(y, 4), 'z': 0.0, 'qx': 0.0, 'qy': 0.0,
                          'qz': round(math.sin(yaw / 2), 6), 'qw': round(math.cos(yaw / 2), 6)}
                         for x, y, yaw in self.points]}
        os.makedirs(os.path.dirname(self.path_file), exist_ok=True)
        with open(self.path_file, 'w', encoding='utf-8') as f:
            yaml.safe_dump(doc, f, allow_unicode=True, sort_keys=False)
        self.get_logger().info(f'已保存 {len(self.points)} 点，长度 {length:.2f} m')
        return True

    def load_path(self):
        with open(self.path_file, 'r', encoding='utf-8') as f:
            doc = yaml.safe_load(f)
        path = Path()
        path.header.frame_id = str(doc.get('frame_id', self.frame_id))
        path.header.stamp = self.get_clock().now().to_msg()
        for it in doc.get('poses', []):
            ps = PoseStamped()
            ps.header = path.header
            ps.pose.position.x = float(it['x'])
            ps.pose.position.y = float(it['y'])
            ps.pose.orientation.z = float(it.get('qz', 0.0))
            ps.pose.orientation.w = float(it.get('qw', 1.0))
            path.poses.append(ps)
        if len(path.poses) < 2:
            raise RuntimeError(f'{self.path_file} 有效路径点不足 2 个')
        self.get_logger().info(f'已加载 {len(path.poses)} 点，长度 '
                               f'{float(doc.get("length_m", 0.0)):.2f} m')
        return path

    def on_feedback(self, msg):
        now = self.get_clock().now().nanoseconds / 1e9
        if now - self.last_fb < 1.0:
            return
        self.last_fb = now
        fb = msg.feedback
        print(f'  剩余 {fb.distance_remaining:.2f} m，恢复 {fb.number_of_recoveries} 次，'
              f'已用 {fb.navigation_time.sec} s', flush=True)

    def cancel(self):
        if self.goal_handle is not None:
            rclpy.spin_until_future_complete(self, self.goal_handle.cancel_goal_async(),
                                             timeout_sec=5.0)

    def play(self):
        client = ActionClient(self, RecordPath, 'navigate_follow_record_path')
        if not client.wait_for_server(timeout_sec=5.0):
            self.get_logger().error('navigate_follow_record_path 不存在，检查 bt_navigator')
            return False
        goal = RecordPath.Goal()
        goal.path = self.path
        goal.behavior_tree = ''     # 空 = 默认 navigate_follow_record_path_w_replanning_and_recovery.xml
        fut = client.send_goal_async(goal, feedback_callback=self.on_feedback)
        rclpy.spin_until_future_complete(self, fut, timeout_sec=10.0)
        self.goal_handle = fut.result()
        if self.goal_handle is None or not self.goal_handle.accepted:
            self.get_logger().error('goal 被拒绝（BT 文件缺失或路径为空）')
            return False
        self.get_logger().info('路径已受理，开始沿示教路径行驶')
        res = self.goal_handle.get_result_async()
        rclpy.spin_until_future_complete(self, res, timeout_sec=self.timeout)
        if not res.done():
            self.get_logger().error(f'{self.timeout:.0f} s 未返回结果，主动取消')
            self.cancel()
            return False
        status = res.result().status
        self.get_logger().info(f'动作结束：{STATUS.get(status, status)}')
        return status == GoalStatus.STATUS_SUCCEEDED


def main(args=None):
    rclpy.init(args=args)
    node = PathTeach()
    ok = False
    try:
        if node.mode == 'record':
            rclpy.spin(node)
        else:
            ok = node.play()
    except KeyboardInterrupt:
        if node.mode != 'record':
            node.cancel()
    finally:
        if node.mode == 'record':
            ok = node.save()
        node.destroy_node()
        if rclpy.ok():
            rclpy.shutdown()
    sys.exit(0 if ok else 1)


if __name__ == '__main__':
    main()
```

## 6. 关键代码与配置

| 代码段 | 作用 | 关键点 |
|---|---|---|
| `yaw_of()` | 四元数 → 偏航角 | 只存 x/y/yaw，回放时重建 `qz/qw` |
| `sample()` | 5 Hz 定时采样 + 抽稀 | AMCL 非定频发布，必须自控采样节奏 |
| `save()` | YAML 落盘 | `sort_keys=False` 保持可读，坐标保留 4 位小数 |
| `load_path()` | YAML → `nav_msgs/Path` | 所有 `PoseStamped.header` 统一为 `map` + 当前时间 |
| `behavior_tree = ''` | 使用默认跟随路径 BT | 源码确认空串回退 `navigate_follow_record_path_w_replanning_and_recovery.xml` |
| `on_feedback()` | 打印剩余距离与恢复次数 | 字段名以 `RecordPath.action` 为准 |
| `cancel()` | 超时/中断取消 | `goal_handle.cancel_goal_async()` |

| 接口 | 类型 | 方向 | 说明 |
|---|---|---|---|
| `/navigate_follow_record_path` | `yhs_can_interfaces/action/RecordPath` | 客户端 | 主通道，可拿 feedback/result |
| `/record_path` | `nav_msgs/msg/Path` | 导航器订阅 | 另一真实入口：外部投喂路径后自动发起 action |
| `/amcl_pose` | `geometry_msgs/msg/PoseWithCovarianceStamped` | 订阅 | 录制源，RELIABLE |

| 参数 | 默认 | 说明 |
|---|---|---|
| `mode` | record | `record` 录制 / `replay` 回放 |
| `path_file` | `~/yhs_labs/S2-09/detour.yaml` | 路径文件 |
| `sample_rate` / `min_spacing` | 5.0 / 0.10 | 采样频率（Hz）/ 位移抽稀（m） |
| `min_yaw` | 0.15 | 转角抽稀（rad） |
| `result_timeout` | 600.0 | 等待 action 结果上限（s） |

## 10. 回滚与常见问题

回滚：①`Ctrl+C` 结束录制/回放，回放自动 `cancel_goal`；②`mv ~/yhs_labs/S2-09/detour.yaml ~/yhs_labs/S2-09/archive_$(date +%m%d).yaml` 归档，或 `rm -rf ~/yhs_labs/S2-09`；③`git status` 应无新增；④不改地图与 Nav2 参数，施工结束后删除临时路径即可。

| 现象 | 原因 | 处理 |
|---|---|---|
| `ModuleNotFoundError: yhs_can_interfaces` | 未 source 工作空间 | `source install/setup.bash` |
| `navigate_follow_record_path 不存在` | `bt_navigator` 未 active | `ros2 lifecycle get /bt_navigator`；重启全栈 |
| `goal 被拒绝（BT 文件缺失）` | 自定义 BT 未安装到 `share/nav2_bt_navigator/behavior_trees/` | 按 S1-07 修正后重试 |
| 车不动却返回 `SUCCEEDED` | 录制时车未移动，路径点重合 | 检查 `length_m`，重新录线 |
| 复现贴墙/切角 | 抽稀过狠 | 把 `min_spacing` 调到 0.05~0.10 m 重录 |
| 起点偏离过大 | 未先到起点 | 人工把车开到起点 0.1 m 内，或先发 `NavigateToPose` |
